\title{Lower bound reasoning with \textsf{eHL}}

\maketitle

\paragraph{The \EHL judgment.}
Let $\Mem$ denote the set of program memories and let
$\mathrm{Exp} = \Mem \to \Real^{+}$ be the set of
non-negative extended-real expectations.
For predicates $\pred, \predtwo : \Mem \to \Bool$,
expectations $\ex, \extwo \in \mathrm{Exp}$, and a program $\cmd$,
the \EHL judgment
\[
  \HT{\pred}{\ex}{\cmd}{\predtwo}{\extwo}
  \qquad {\rel} \in \{{\leq},=,{\geq}\}
\]

\paragraph{Validity.}
Let $\cmd_\mem$ denote the output distribution of $\cmd$ from memory $\mem$.
A judgment $\valid \HT{\pred}{\ex}{\cmd}{\predtwo}{\extwo}$ is \emph{valid}
if both of the following hold:
\begin{enumerate}
  \item $\forall\mem \in \Mem . \pred(\mem) \implies \supp(\cmd_\mem) \subseteq \predtwo$
  \item $\forall\mem \in \Mem . \pred(\mem) \implies \ex(\mem) \;\HTrel\; \E{\cmd,\mem}{[\predtwo]\cdot\extwo}$
\end{enumerate}
Condition~(1) requires that all output memories satisfy the postcondition
$\predtwo$. Condition~(2) requires that $\ex(\mem)$ bounds the expected
value of $\extwo$ restricted to outputs satisfying $\predtwo$.

\paragraph{Remark.}\label{rem:validity-indicator}
Since Condition~(1) guarantees $\supp(\cmd_\mem) \subseteq \predtwo$,
the indicator $[\predtwo]$ evaluates to $1$ on every output state of $\cmd$, and therefore
\[
  \E{\cmd,\mem}{[\predtwo]\cdot\extwo} \;=\; \E{\cmd,\mem}{\extwo}.
\]

\paragraph{Notation.}
For a predicate $\pred : \Mem \to \Bool$ we use two indicator-style operators, one tailored to
each direction of reasoning:
\[
  [\pred](\mem) \;=\;
  \begin{cases} 1 & \text{if } \pred(\mem) \\ 0 & \text{otherwise} \end{cases}
  \qquad\text{(used for lower bounds)}
  \\[3mm]
\]
\[
  \langle \pred \rangle(\mem) \;=\;
  \begin{cases} 1 & \text{if } \pred(\mem) \\ \infty & \text{otherwise} \end{cases}
  \qquad\text{(used for upper bounds)}
  \\[3mm]
\]
% Given a memory-indexed expectation $r : \Mem \to \Real^{+}$,
% the expressions $[\pred] \cdot r$ and $\langle \pred \rangle \cdot r$ denote
% the corresponding pointwise products over $\Mem$:
% $[\pred] \cdot r$ agrees with $r$ on memories satisfying $\pred$ and is zero
% elsewhere, while $\langle \pred \rangle \cdot r$ agrees with $r$ on memories
% satisfying $\pred$ and is $\infty$ elsewhere.

\paragraph{The witness sequence $\{r_i\}_{i \in \mathbb{N}}$.}
The rule of \Cref{fig:hoare:while-lb} uses a sequence $\{r_i\}_{i \geq 0}$ of
non-negative expectations as a termination witness: intuitively, $r_i$ is a
(user-chosen) upper bound on how much of the post-expectation mass of $\ex$ is
still trapped inside the loop after $i$ body executions. Premise~2 ties $r_i$
to $\ex$ at $i=0$ by requiring $r_0 \geq \ex$; Premise~3 propagates the bound
across one iteration of the body; and Premise~4 forces the bound --- and thus
the trapped mass --- to vanish in the limit.

        & \forall i \geq 0.\, \HT[upper]{\bexpr\land\pred}{r_{i+1}}{\cmd}{\top}{r_i}

\end{figure}

\section{Soundness and Completeness}\label{app:soundness-completeness}
% ======================================================================

\subsection{Foundations}\label{app:foundations}

This subsection makes explicit the typed semantic objects used by both the
soundness and the completeness arguments below.

\paragraph{Memories, predicates, and expectations.}
Let $\Mem$ be the set of program memories.  Predicates and expectations have
different codomains:
\begin{equation}
  P,Q:\Mem\to\Bool,
  \qquad
  f,g:\Mem\to[0,\infty].
  \label{eq:cook-semantic-types}
\end{equation}
All comparisons between expectations are pointwise; for example,
$f\leq g$ abbreviates $\forall m\in\Mem.\ f(m)\leq g(m)$.

\paragraph{Commands and distribution semantics.}
The command language considered below is
\begin{equation}
  C ::= \SKIP
  \mid \vx <- \expr
  \mid \vx <* \sexpr
  \mid C_1;C_2
  \mid \ITE{B}{C_1}{C_2}
  \mid \WHILE B\DO D.
  \label{eq:cook-command-syntax}
\end{equation}
Every command denotes a subdistribution-valued state transformer
\begin{equation}
  \llbracket C\rrbracket:
  \Mem\longrightarrow\mathcal D_{\leq1}(\Mem).
  \label{eq:cook-command-semantics-type}
\end{equation}
Subdistributions are used because non-termination may lose probability mass.
Two operations build every distribution used below, based on Avanzini
et al.\ [2024], Section~4. For $v\in\Mem$, write $\delta_v$ for
the \emph{point (Dirac) distribution} at $v$,
\begin{equation}
  \delta_v(m) \;\defsym\;
  \begin{cases} 1 & m=v, \\ 0 & \text{otherwise,}\end{cases}
\end{equation}
and define $\operatorname{return}(v)\in\mathcal D_{\leq1}(\Mem)$ by
\begin{equation}
  \operatorname{return}(v) \;\defsym\; \delta_v.
  \label{eq:cook-return-def}
\end{equation}
Given $\mu\in\mathcal D_{\leq1}(\Mem)$ and a family
$K:\Mem\to\mathcal D_{\leq1}(\Mem)$, the \emph{(monadic) sequencing}
operation $\operatorname{bind}(\mu,K)\in\mathcal D_{\leq1}(\Mem)$ mixes the
continuations $K(x)$ according to $\mu$:
\begin{equation}
  \operatorname{bind}(\mu,K)(m) \;\defsym\;
  \sum_{x\in\operatorname{supp}(\mu)} \mu(x)\cdot K(x)(m).
  \label{eq:cook-bind-def}
\end{equation}
With these two operations, the clauses needed most directly below include
\begin{align}
  \llbracket\SKIP\rrbracket(m)
    &=\operatorname{return}(m),
  \label{eq:cook-denote-skip}\\
  \llbracket\vx <- \expr\rrbracket(m)
    &=\operatorname{return}\bigl(m[\vx\mapped\expr(m)]\bigr),
  \label{eq:cook-denote-assign}\\
  \llbracket\vx <* \sexpr\rrbracket(m)
    &=\operatorname{bind}\bigl(\sexpr(m),
       \lambda v.\operatorname{return}(m[\vx\mapped v])\bigr),
  \label{eq:cook-denote-sample}\\
  \llbracket C_1;C_2\rrbracket(m)
    &=\operatorname{bind}\bigl(\llbracket C_1\rrbracket(m),
       \lambda m'.\llbracket C_2\rrbracket(m')\bigr).
  \label{eq:cook-denote-seq}\\
  \llbracket\ITE{B}{C_1}{C_2}\rrbracket(m)
    &=
      \begin{cases}
        \llbracket C_1\rrbracket(m),&B(m),\\
        \llbracket C_2\rrbracket(m),&\neg B(m).
      \end{cases}
  \label{eq:cook-denote-if}
\end{align}
For the loop, let $X:\Mem\to\mathcal D_{\leq1}(\Mem)$ and define the kernel
functional
\begin{equation}
  \mathcal F_{B,D}(X)(m)
  \defsym
  \begin{cases}
    \operatorname{bind}(\llbracket D\rrbracket(m),X),&B(m),\\
    \operatorname{return}(m),&\neg B(m).
  \end{cases} 
  \label{eq:cook-denote-loop-functional}
\end{equation}
Then
\begin{equation}
  \llbracket\WHILE B\DO D\rrbracket
  \defsym\operatorname{lfp}(\mathcal F_{B,D})
  \label{eq:cook-denote-loop}
\end{equation}
in the pointwise order on subdistribution kernels.

For a subdistribution $\mu$ and a non-negative function $h$, write
$\mathbb E_{x\sim\mu}[h(x)]\defsym\sum_{x\in\Mem}\mu(x)h(x)$ for its
expectation. Two elementary laws, used repeatedly below, follow directly
from the definitions of $\operatorname{return}$
(\ref{eq:cook-return-def}) and $\operatorname{bind}$
(\ref{eq:cook-bind-def}). The first is immediate, since
$\operatorname{return}(v)$ puts all its mass on $v$:
\begin{equation}
  \mathbb E_{x\sim\operatorname{return}(v)}[h(x)]
    =\sum_{x\in\Mem}\operatorname{return}(v)(x)\,h(x)
    =h(v).
  \label{eq:cook-expect-return}
\end{equation}
The second follows by unfolding $\operatorname{bind}$ and swapping the
order of summation:
\begin{align}
  \mathbb E_{y\sim\operatorname{bind}(\mu,K)}[h(y)]
    &=\sum_{y\in\Mem}\operatorname{bind}(\mu,K)(y)\,h(y)
    =\sum_{y\in\Mem}\sum_{x\in\Mem}\mu(x)\,K(x)(y)\,h(y)\\
    &=\sum_{x\in\Mem}\mu(x)\sum_{y\in\Mem}K(x)(y)\,h(y)
    =\mathbb E_{x\sim\mu}
       \left[\mathbb E_{y\sim K(x)}[h(y)]\right].
  \label{eq:cook-expect-bind}
\end{align}
The second equation is the law of iterated expectation for sequential
composition; all sums involved are of non-negative terms, so the
rearrangement is justified regardless of convergence.

\paragraph{The weakest-precondition operator.}
A single operator $\WP{C}{\cdot}$ is used throughout, applied either to an
expectation or to a predicate; which of the two defining equations applies
is determined by the type of the argument --- no separate notation is
introduced for the two cases:
\begin{align}
  \WP{C}{g}(m)
    &\defsym
      \mathbb E_{m'\sim\llbracket C\rrbracket(m)}[g(m')],
    &&g:\Mem\to[0,\infty],
  \label{eq:cook-wp-definition}\\
  \WP{C}{Q}(m)
    &\defsym
      \forall m'\in\supp(\llbracket C\rrbracket(m)).\ Q(m'),
    &&Q:\Mem\to\Bool.
  \label{eq:cook-gwp-definition}
\end{align}
Applied to an expectation, $\WP{C}{g}$ is quantitative: it returns the exact
expected value of $g$ after running $C$. Applied to a predicate, $\WP{C}{Q}$
is the ordinary (liberal) weakest precondition, expressing that every
reachable output state satisfies $Q$.

The return and bind laws give the corresponding equations for the predicate
case,
\begin{equation}
  \WP{\SKIP}{Q}=Q,
  \qquad
  \WP{C_1;C_2}{Q}
  =\WP{C_1}{\WP{C_2}{Q}}.
  \label{eq:cook-gwp-basic-laws}
\end{equation}
The identical-looking equations for the expectation case are
\eqref{eq:cook-wp-skip} and \eqref{eq:cook-wp-seq} below; both pairs are
instances of the same return/bind laws, one for each type of argument.

\paragraph{Semantic validity versus syntactic derivability.}
For $\bowtie\in\{\leq,=,\geq\}$, semantic validity of a guarded judgment is
\begin{equation}
  \valid\HT{P}{f}{C}{Q}{g}
  \quad\Longleftrightarrow\quad
  \forall m\in\Mem.\ P(m)\Longrightarrow{}
  \bigl(\WP{C}{Q}(m) \land f(m)\bowtie\WP{C}{g}(m)\bigr).
  \label{eq:cook-validity-typed}
\end{equation}
The first conjunct is qualitative: every output state in the support
satisfies $Q$. The second is quantitative: $f$ has the required relation to
the exact expected value of $g$. The two conjuncts are not independent: since
the first conjunct forces $[Q]=1$ on the whole output support, it gives
$\mathbb E[[Q]\cdot g]=\mathbb E[g]$, so \eqref{eq:cook-validity-typed} is
equivalent to the validity definition used at the start of the paper, which
weights $g$ by $[Q]$ explicitly instead of stating support-inclusion as a
separate conjunct. Throughout the rest of this section, $\vDash$ denotes
semantic truth and $\vdash$ denotes derivability from the displayed proof
rules.

\subsection{Soundness of the While Rule}\label{app:while-soundness}

\paragraph{Goal.}
We want to show that the \textsc{While-lb} rule of \Cref{fig:hoare:while-lb}
is sound, i.e.\ that its conclusion
$\HT[lower]{\pred}{\ex}{\WHILE\bexpr\DO\cmd}{\neg\bexpr\land\pred}{\ex}$ is
valid. Its postcondition is $\neg\bexpr\land\pred$, so by validity
Condition~(2) the indicator $[\pred\land\neg\bexpr]$ below is exactly the
postcondition indicator applied to $\ex$; we keep it explicit because it is
manipulated directly throughout the proof:
\[
  [\pred] \cdot \ex
  \leq
  \WP{\WHILE\bexpr\DO\cmd}{[\pred \wedge \neg\bexpr] \cdot \ex}
\]

\paragraph{Definitions.}
Define the family of approximating programs:
\[
  W_n
  \;\defsym\;
  \bigl(\kwd{if}\ \bexpr\ \kwd{then}\ \cmd\bigr)^n
  \mathbin{;}
  \kwd{if}\ \bexpr\ \kwd{then}\ \ABORT
\]
where $W_n$ allows at most $n$ iterations; if the loop would continue beyond $n$,
it aborts. Define:
\[
  I_n \;\defsym\; \WP{W_n}{[\pred \wedge \neg\bexpr] \cdot \ex}
\]

% ----------------------------------------------------------------------
\begin{lemma}[Recursive characterisation of $I_n$]\label{lem:In-recursion}
\[
  I_0 \;=\; [\pred \wedge \neg\bexpr] \cdot \ex
  \qquad\text{and}\qquad
  I_{n+1}
  \;=\; [\bexpr] \cdot \WP{\cmd}{I_n}
        + [\pred \wedge \neg\bexpr] \cdot \ex
\]
\end{lemma}

\begin{proof}
By induction on $n$.

\paragraph{Base case $n=0$.}
\begin{align*}
  I_0
  &= \WP{\kwd{if}\ \bexpr\ \kwd{then}\ \ABORT}{[\pred \wedge \neg\bexpr] \cdot \ex} \\
  &= [\bexpr] \cdot \WP{\ABORT}{[\pred \wedge \neg\bexpr] \cdot \ex}
     + [\neg\bexpr] \cdot [\pred \wedge \neg\bexpr] \cdot \ex \\
  &= [\bexpr] \cdot 0 + [\pred \wedge \neg\bexpr] \cdot \ex
   \;=\; [\pred \wedge \neg\bexpr] \cdot \ex
\end{align*}
where we used $\WP{\ABORT}{\cdot} = 0$ and
$[\neg\bexpr] \cdot [\pred \wedge \neg\bexpr] = [\pred \wedge \neg\bexpr]$.

\paragraph{Inductive step $n \to n+1$.}
Since $(\kwd{if}\ \bexpr\ \kwd{then}\ \cmd)^{n+1}
 = \kwd{if}\ \bexpr\ \kwd{then}\ \cmd
   \mathbin{;} (\kwd{if}\ \bexpr\ \kwd{then}\ \cmd)^n$, we have
$W_{n+1} = (\kwd{if}\ \bexpr\ \kwd{then}\ \cmd) \mathbin{;} W_n$, so:
\begin{align*}
  I_{n+1}
  &= \WP{\kwd{if}\ \bexpr\ \kwd{then}\ \cmd}{\WP{W_n}{[\pred \wedge \neg\bexpr] \cdot \ex}} \\
  &= \WP{\kwd{if}\ \bexpr\ \kwd{then}\ \cmd}{I_n} \\
  &= [\bexpr] \cdot \WP{\cmd}{I_n} + [\neg\bexpr] \cdot I_n
\end{align*}
It remains to identify $[\neg\bexpr] \cdot I_n$. When $\neg\bexpr$ already
holds at the starting memory, none of the guard checks inside $W_n$ ever
fire, so the memory is left untouched and $I_n$ reduces directly to
$[\pred \wedge \neg\bexpr] \cdot \ex$:
\[
  [\neg\bexpr] \cdot I_n
  \;=\; [\pred \wedge \neg\bexpr] \cdot \ex.
\]
Therefore:
\[
  I_{n+1}
  \;=\; [\bexpr] \cdot \WP{\cmd}{I_n} + [\pred \wedge \neg\bexpr] \cdot \ex
\]
\end{proof}

% ----------------------------------------------------------------------

\begin{lemma}[Key inequality]\label{lem:key-ineq}
Assuming premises \textnormal{(a)}, \textnormal{(b)}, \textnormal{(c)},
for all $n \geq 0$:
\[
  [\pred] \cdot I_n + [\pred] \cdot r_n \;\geq\; [\pred] \cdot \ex
\]
\end{lemma}

\begin{proof}
By induction on $n$.

\paragraph{Base case $n = 0$.}
\[
  [\pred] \cdot I_0 + [\pred] \cdot r_0
  \;=\; [\pred \wedge \neg\bexpr] \cdot \ex + [\pred] \cdot r_0
\]
By premise~(b), $[\pred] \cdot r_0 \geq [\pred \wedge \bexpr] \cdot \ex$, so:
\[
  \;\geq\; [\pred \wedge \neg\bexpr] \cdot \ex + [\pred \wedge \bexpr] \cdot \ex
  \;=\; [\pred] \cdot \ex
\]

\paragraph{Inductive step $n \to n+1$.}
Expanding $I_{n+1}$ by \Cref{lem:In-recursion} and multiplying by $[\pred]$:
\begin{align*}
  [\pred] \cdot I_{n+1} + [\pred] \cdot r_{n+1}
  &= [\pred \wedge \bexpr] \cdot \WP{\cmd}{I_n}
     + [\pred \wedge \neg\bexpr] \cdot \ex
     + [\pred] \cdot r_{n+1}
\end{align*}
Premise~(c) gives $[\pred \wedge \bexpr] \cdot r_{n+1} \geq \WP{\cmd}{r_n}$. Multiplying by
$[\pred]$ (using $[\pred]\cdot[\pred\wedge\bexpr] = [\pred\wedge\bexpr]$):
\[
  [\pred \wedge \bexpr] \cdot r_{n+1} \;\geq\; [\pred] \cdot \WP{\cmd}{r_n}
\]
Since $r_n \geq [\pred]\cdot r_n$ pointwise ($r_n$ is non-negative), monotonicity of $\WP{\cmd}{\cdot}$
gives $\WP{\cmd}{r_n} \geq \WP{\cmd}{[\pred]\cdot r_n}$, hence
$[\pred]\cdot\WP{\cmd}{r_n} \geq [\pred\wedge\bexpr]\cdot\WP{\cmd}{[\pred]\cdot r_n}$. Combining, and
extending from $[\pred\wedge\bexpr]$ to $[\pred]$ using non-negativity of $r_{n+1}$ on the
$[\pred\wedge\neg\bexpr]$ part:
\[
  [\pred] \cdot r_{n+1} \;\geq\; [\pred \wedge \bexpr] \cdot \WP{\cmd}{[\pred] \cdot r_n}
\]
Substituting into the expansion of $[\pred]\cdot I_{n+1}+[\pred]\cdot r_{n+1}$ above:
\begin{align*}
  [\pred] \cdot I_{n+1} + [\pred] \cdot r_{n+1}
  &\geq [\pred \wedge \bexpr] \cdot \WP{\cmd}{I_n}
        + [\pred \wedge \bexpr] \cdot \WP{\cmd}{[\pred] \cdot r_n}
        + [\pred \wedge \neg\bexpr] \cdot \ex
\end{align*}
By linearity of $\WP{\cmd}{\cdot}$:
\begin{align*}
  &= [\pred \wedge \bexpr] \cdot \WP{\cmd}{I_n + [\pred] \cdot r_n}
     + [\pred \wedge \neg\bexpr] \cdot \ex
\end{align*}
Since $I_n \geq [\pred]\cdot I_n$ pointwise ($I_n$ is non-negative), monotonicity of $\WP{\cmd}{\cdot}$
gives:
\[
  [\pred \wedge \bexpr] \cdot \WP{\cmd}{I_n + [\pred] \cdot r_n}
  \;\geq\; [\pred \wedge \bexpr] \cdot \WP{\cmd}{[\pred]\cdot I_n + [\pred] \cdot r_n}
\]
By the induction hypothesis
$[\pred] \cdot I_n + [\pred] \cdot r_n \geq [\pred] \cdot \ex$
and monotonicity of $\WP{\cmd}{\cdot}$:
\begin{align*}
  &\geq [\pred \wedge \bexpr] \cdot \WP{\cmd}{[\pred] \cdot \ex}
        + [\pred \wedge \neg\bexpr] \cdot \ex
\end{align*}
By premise~(a), validity Condition~(2) gives
$[\pred \wedge \bexpr] \cdot \WP{\cmd}{[\pred] \cdot \ex} \geq [\pred \wedge \bexpr] \cdot \ex$:
\[
  \;\geq\; [\pred \wedge \bexpr] \cdot \ex + [\pred \wedge \neg\bexpr] \cdot \ex
  \;=\; [\pred] \cdot \ex
\]
\end{proof}

% ----------------------------------------------------------------------
\paragraph{Main result.}
From \Cref{lem:key-ineq}, for all $n \geq 0$:
\[
  [\pred] \cdot I_n + [\pred] \cdot r_n \;\geq\; [\pred] \cdot \ex
\]
Taking the limit:
\[
  \lim_n ([\pred] \cdot I_n) + \lim_n ([\pred] \cdot r_n) \;\geq\; [\pred] \cdot \ex
\]
By condition~(d), $\lim_n r_n = 0$, so $\lim_n ([\pred] \cdot r_n) = 0$:
\[
  \lim_n ([\pred] \cdot I_n) \;\geq\; [\pred] \cdot \ex
\]
By definition and \Cref{lem:In-recursion},
$\lim_n I_n = \WP{\WHILE\bexpr\DO\cmd}{[\pred \wedge \neg\bexpr] \cdot \ex}$,
giving:
\[
  \WP{\WHILE\bexpr\DO\cmd}{[\pred \wedge \neg\bexpr] \cdot \ex}
  \;\geq\; [\pred] \cdot \ex
\]
which is exactly the desired conclusion. \qed

\subsection{Syntax and Semantics}\label{app:lj-semantics}

\paragraph{Syntax.}
The command language extends \eqref{eq:cook-command-syntax} with jumps
and labeled blocks:
\begin{equation}
  C ::= \SKIP
  \mid \vx <- \expression{e}
  \mid \vx <* \sexpr
  \mid C_1;C_2
  \mid \ITE{\bexpr}{C_1}{C_2}
  \mid \WHILE \bexpr\DO C
  \mid \mathsf{jump}\ \ell
  \mid \ell\{C\}.
  \label{eq:lj-syntax}
\end{equation}
\todo{TODO: semantics, equations, rule and soundness for $\mathsf{while}$.}

\paragraph{Results.}
\[
  \ell\in\mathsf{Lab},
  \qquad
  \mathsf{Res}\ a::=\mathsf{Ok}(a)\mid\mathsf{Jump}(\ell,m),
\]
\[
  \mathsf{Ok}:a\to\mathsf{Res}\ a,
  \qquad
  \mathsf{Jump}:\mathsf{Lab}\to\Mem\to\mathsf{Res}\ a,
  \qquad
  \llbracket C\rrbracket:\Mem\to\mathcal D_{\leq1}(\mathsf{Res}\ \Mem).
\]
The missing probability mass represents non-termination.

\paragraph{Semantics.}
For a distribution expression $\sexpr$, $\sexpr(m)$ is a subdistribution
over values.
\begin{align}
  \llbracket\SKIP\rrbracket(m)
  &=\operatorname{return}\bigl(\mathsf{Ok}(m)\bigr),
  \label{eq:lj-sem-skip}\\
  \llbracket \vx <- \expression{e}\rrbracket(m)
  &=\operatorname{return}
    \bigl(\mathsf{Ok}(m[\vx\mapped\expression{e}(m)])\bigr),
  \label{eq:lj-sem-assign}\\
  \llbracket \vx <* \sexpr\rrbracket(m)
  &=\operatorname{bind}\bigl(\sexpr(m),\
    \lambda v.\,\operatorname{return}(\mathsf{Ok}(m[\vx\mapped v]))\bigr),
  \label{eq:lj-sem-sample}\\
  \llbracket\ITE{\bexpr}{C_1}{C_2}\rrbracket(m)
  &=\begin{cases}
      \llbracket C_1\rrbracket(m),&\bexpr(m),\\
      \llbracket C_2\rrbracket(m),&\neg\bexpr(m),
    \end{cases}
  \label{eq:lj-sem-if}\\
  \llbracket\mathsf{jump}\ \ell\rrbracket(m)
  &=\operatorname{return}\bigl(\mathsf{Jump}(\ell,m)\bigr).
  \label{eq:lj-sem-jump}
\end{align}
For $\mu:\mathcal D_{\leq1}(\mathsf{Res}\ a)$ and
$k:a\to\mathcal D_{\leq1}(\mathsf{Res}\ b)$, the bind of the result monad
runs $k$ on a normal result and passes a jump through unchanged:
\begin{equation}
  \mathrm{dlet}_{\mathrm{RM}}\ x\leftarrow\mu\ \mathrm{in}\ k(x)
  \defsym
  \operatorname{bind}\left(\mu,\
  \lambda r.\,
  \begin{cases}
    k(a'),&r=\mathsf{Ok}(a'),\\
    \operatorname{return}(\mathsf{Jump}(\ell,m')),
      &r=\mathsf{Jump}(\ell,m')
  \end{cases}\right).
  \label{eq:lj-dlet}
\end{equation}
\begin{equation}
  \llbracket C;D\rrbracket(m)
  =\mathrm{dlet}_{\mathrm{RM}}\ m'\leftarrow\llbracket C\rrbracket(m)\
  \mathrm{in}\ \llbracket D\rrbracket(m').
  \label{eq:lj-sem-seq}
\end{equation}
\begin{equation}
  \llbracket\ell\{C\}\rrbracket(m)
  =\operatorname{bind}\left(\llbracket C\rrbracket(m),\
  \lambda r.\,
  \begin{cases}
    \operatorname{return}(\mathsf{Ok}(m')),
      &r=\mathsf{Ok}(m'),\\
    \operatorname{return}(\mathsf{Ok}(m')),
      &r=\mathsf{Jump}(\ell,m'),\\
    \operatorname{return}(\mathsf{Jump}(\ell',m')),
      &r=\mathsf{Jump}(\ell',m'),\ \ell'\neq\ell
  \end{cases}\right)
  \label{eq:lj-sem-label}
\end{equation}

\subsection{Weakest Pre-Expectation and Weakest Guard}\label{app:lj-wp}

\paragraph{Label environments.}
\[
  \mathsf{Pred}\defsym\Mem\to\Bool,
  \qquad
  \mathsf{Exp}\defsym\Mem\to[0,\infty].
\]
\[
  G:\mathsf{Lab}\to\mathsf{Pred},
  \qquad
  E:\mathsf{Lab}\to\mathsf{Exp}.
\]
The notation $G\mid E$ pairs the predicate and expectation
environments, as in a guarded assertion $\{P\mid f\}$.
Updating an environment at a label $\ell$ is written
\begin{align}
  E[\ell\mapsto g](\ell')
  &\defsym
  \begin{cases}
    g,&\ell'=\ell,\\
    E(\ell'),&\ell'\neq\ell,
  \end{cases}
  \nonumber\\
  G[\ell\mapsto Q](\ell')
  &\defsym
  \begin{cases}
    Q,&\ell'=\ell,\\
    G(\ell'),&\ell'\neq\ell.
  \end{cases}
  \label{eq:lj-envupd}
\end{align}

\paragraph{Transformers.}
For a post-expectation $g:\mathsf{Exp}$ and a postcondition
$Q:\mathsf{Pred}$, the functions
$(E;\mathsf{Ok}\mapsto g):\mathsf{Res}\ \Mem\to[0,\infty]$ and
$(G;\mathsf{Ok}\mapsto Q):\mathsf{Res}\ \Mem\to\Bool$ read off a result with
$g$ (resp.\ $Q$) on a normal exit and with the environment on a jump:
\begin{align}
  (E;\mathsf{Ok}\mapsto g)(r)
  &\defsym
  \begin{cases}
    g(m'),&r=\mathsf{Ok}(m'),\\
    E(\ell)(m'),&r=\mathsf{Jump}(\ell,m'),
  \end{cases}
  \nonumber\\
  (G;\mathsf{Ok}\mapsto Q)(r)
  &\defsym
  \begin{cases}
    Q(m'),&r=\mathsf{Ok}(m'),\\
    G(\ell)(m'),&r=\mathsf{Jump}(\ell,m').
  \end{cases}
  \label{eq:ljwp-okmap}
\end{align}
The two transformers $\mathrm{ewp}_{E}[C]:\mathsf{Exp}\to\mathsf{Exp}$
and $\mathrm{wp}_{G}[C]:\mathsf{Pred}\to\mathsf{Pred}$ are
\begin{equation}
  \mathrm{ewp}_{E}[C](g)(m)
  \defsym
  \mathbb E_{r\sim\llbracket C\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr],
  \label{eq:ljwp-wp}
\end{equation}
\begin{equation}
  \mathrm{wp}_{G}[C](Q)(m)
  \defsym
  \forall r\in\supp(\llbracket C\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r).
  \label{eq:ljwp-wpG}
\end{equation}
\paragraph{Semantic validity.}
\begin{equation}
  \begin{aligned}
  &\models G\mid E\vdash\{P\mid f\}_{\bowtie}\ C\ \{Q\mid g\}\\
  &\qquad\iff
  \forall m.\ P(m)\Longrightarrow
  \mathrm{wp}_{G}[C](Q)(m)
  \ \land\
  f(m)\mathrel{\bowtie}\mathrm{ewp}_{E}[C](g)(m).
  \end{aligned}
  \label{eq:ljwp-validity}
\end{equation}

\subsection{Monad Facts}\label{app:lj-monad}

For sets $X,Y$, $v\in X$, $\mu\in\mathcal D_{\leq1}(X)$,
$K:X\to\mathcal D_{\leq1}(Y)$, $h:X\to[0,\infty]$, $h':Y\to[0,\infty]$,
$\varphi:X\to\Bool$ and $\varphi':Y\to\Bool$:
\begin{equation}
  \begin{aligned}
  \mathbb E_{r\sim\operatorname{return}(v)}[h(r)]&=h(v),\\
  \bigl(\forall r\in\supp(\operatorname{return}(v)).\ \varphi(r)\bigr)
  &\iff\varphi(v),
  \end{aligned}
  \label{eq:lj-return}
\end{equation}
\begin{equation}
  \begin{aligned}
  \mathbb E_{s\sim\operatorname{bind}(\mu,K)}[h'(s)]
  &=\mathbb E_{r\sim\mu}\bigl[\mathbb E_{s\sim K(r)}[h'(s)]\bigr],\\
  \bigl(\forall s\in\supp(\operatorname{bind}(\mu,K)).\ \varphi'(s)\bigr)
  &\iff
  \forall r\in\supp(\mu).\ \forall s\in\supp(K(r)).\ \varphi'(s).
  \end{aligned}
  \label{eq:lj-bind}
\end{equation}
The expectation equations are \eqref{eq:cook-expect-return} and
\eqref{eq:cook-expect-bind}, whose derivations do not depend on $\Mem$.
For the support equivalences, recall $\supp(\mu)=\{x\mid\mu(x)>0\}$, the
set of outcomes that can actually occur. Since $\operatorname{return}(v)$
puts all its mass on $v$, we have $\supp(\operatorname{return}(v))=\{v\}$.
For $\operatorname{bind}(\mu,K)$, an outcome $s$ can occur exactly when
some first outcome $r$ of $\mu$ can occur and $s$ can occur after it:
\[
  s\in\supp(\operatorname{bind}(\mu,K))
  \iff
  \exists r\in\supp(\mu).\ s\in\supp(K(r)).
\]
Indeed, $\operatorname{bind}(\mu,K)(s)=\sum_{r}\mu(r)\,K(r)(s)$ is a sum of
non-negative numbers, so it is positive iff one of its terms
$\mu(r)\,K(r)(s)$ is positive, i.e.\ iff $\mu(r)>0$ and $K(r)(s)>0$ for
some $r$. Consequently, a property holds for every outcome of
$\operatorname{bind}(\mu,K)$ iff, for every outcome $r$ of $\mu$, it holds
for every outcome of $K(r)$, which is the second line of
\eqref{eq:lj-bind}.

\begin{lemma}[$\mathrm{dlet}_{\mathrm{RM}}$]\label{lem:lj-dlet}
For all $E,G$, $g:\mathsf{Exp}$, $Q:\mathsf{Pred}$,
$\mu:\mathcal D_{\leq1}(\mathsf{Res}\ \Mem)$ and
$k:\Mem\to\mathcal D_{\leq1}(\mathsf{Res}\ \Mem)$:
\begin{align}
  &\mathbb E_{s\sim\mathrm{dlet}_{\mathrm{RM}}\ x\leftarrow\mu\ \mathrm{in}\ k(x)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]
  \nonumber\\
  &\qquad=\mathbb E_{r\sim\mu}
  \Bigl[\Bigl(E;\ \mathsf{Ok}\mapsto\lambda x.\
  \mathbb E_{s\sim k(x)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr)(r)\Bigr],
  \label{eq:lj-dlet-exp}\\
  &\forall s\in\supp\bigl(\mathrm{dlet}_{\mathrm{RM}}\ x\leftarrow\mu\ \mathrm{in}\ k(x)\bigr).\
  (G;\mathsf{Ok}\mapsto Q)(s)
  \nonumber\\
  &\qquad\iff\forall r\in\supp(\mu).\
  \Bigl(G;\ \mathsf{Ok}\mapsto\lambda x.\
  \forall s\in\supp(k(x)).\ (G;\mathsf{Ok}\mapsto Q)(s)\Bigr)(r).
  \label{eq:lj-dlet-guard}
\end{align}
\end{lemma}
\begin{proof}
Let $K$ be the function of $r$ in \eqref{eq:lj-dlet}, so that
$\mathrm{dlet}_{\mathrm{RM}}\ x\leftarrow\mu\ \mathrm{in}\ k(x)=\operatorname{bind}(\mu,K)$.
For the expectation,
\[
  \mathbb E_{s\sim\operatorname{bind}(\mu,K)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]
  \overset{\eqref{eq:lj-bind}}{=}
  \mathbb E_{r\sim\mu}\Bigl[\mathbb E_{s\sim K(r)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr],
\]
and it remains to show, for every $r$,
\[
  \mathbb E_{s\sim K(r)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]
  =\Bigl(E;\ \mathsf{Ok}\mapsto\lambda x.\
  \mathbb E_{s\sim k(x)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr)(r).
\]
If $r=\mathsf{Ok}(m')$:
\[
  \begin{aligned}
  &\mathbb E_{s\sim K(\mathsf{Ok}(m'))}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-dlet}}{=}
  \mathbb E_{s\sim k(m')}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  \Bigl(E;\ \mathsf{Ok}\mapsto\lambda x.\
  \mathbb E_{s\sim k(x)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr)(\mathsf{Ok}(m')).
  \end{aligned}
\]
If $r=\mathsf{Jump}(\ell,m')$:
\[
  \begin{aligned}
  &\mathbb E_{s\sim K(\mathsf{Jump}(\ell,m'))}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-dlet}}{=}
  \mathbb E_{s\sim\operatorname{return}(\mathsf{Jump}(\ell,m'))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)(\mathsf{Jump}(\ell,m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  E(\ell)(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  \Bigl(E;\ \mathsf{Ok}\mapsto\lambda x.\
  \mathbb E_{s\sim k(x)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr)(\mathsf{Jump}(\ell,m')).
  \end{aligned}
\]
For the guard,
\[
  \forall s\in\supp(\operatorname{bind}(\mu,K)).\ (G;\mathsf{Ok}\mapsto Q)(s)
  \overset{\eqref{eq:lj-bind}}{\iff}
  \forall r\in\supp(\mu).\ \forall s\in\supp(K(r)).\ (G;\mathsf{Ok}\mapsto Q)(s),
\]
and it remains to show, for every $r$,
\[
  \forall s\in\supp(K(r)).\ (G;\mathsf{Ok}\mapsto Q)(s)
  \iff\Bigl(G;\ \mathsf{Ok}\mapsto\lambda x.\
  \forall s\in\supp(k(x)).\ (G;\mathsf{Ok}\mapsto Q)(s)\Bigr)(r).
\]
If $r=\mathsf{Ok}(m')$:
\[
  \begin{aligned}
  &\forall s\in\supp(K(\mathsf{Ok}(m'))).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-dlet}}{\iff}
  \forall s\in\supp(k(m')).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  \Bigl(G;\ \mathsf{Ok}\mapsto\lambda x.\
  \forall s\in\supp(k(x)).\ (G;\mathsf{Ok}\mapsto Q)(s)\Bigr)(\mathsf{Ok}(m')).
  \end{aligned}
\]
If $r=\mathsf{Jump}(\ell,m')$:
\[
  \begin{aligned}
  &\forall s\in\supp(K(\mathsf{Jump}(\ell,m'))).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-dlet}}{\iff}
  \forall s\in\supp(\operatorname{return}(\mathsf{Jump}(\ell,m'))).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)(\mathsf{Jump}(\ell,m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  G(\ell)(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  \Bigl(G;\ \mathsf{Ok}\mapsto\lambda x.\
  \forall s\in\supp(k(x)).\ (G;\mathsf{Ok}\mapsto Q)(s)\Bigr)(\mathsf{Jump}(\ell,m')).
  \end{aligned}
\]
\end{proof}

\subsection{Basic Equations}\label{app:lj-equations}

\begin{lemma}[Basic equations]\label{lem:ljwp-equations}
For all $E,G,g,Q$, memories $m$, commands $C,D,C_1,C_2$, labels $\ell$,
Boolean expressions $\bexpr$ and distribution expressions $\sexpr$:
\begin{align}
  \mathrm{ewp}_{E}[\SKIP](g)&=g,
  \nonumber\\
  \mathrm{wp}_{G}[\SKIP](Q)&=Q,
  \label{eq:ljwp-skip}\\[1ex]
  \mathrm{ewp}_{E}[\vx <- \expression{e}](g)(m)
  &=g(m[\vx\mapped\expression{e}(m)]),
  \nonumber\\
  \mathrm{wp}_{G}[\vx <- \expression{e}](Q)(m)
  &=Q(m[\vx\mapped\expression{e}(m)]),
  \label{eq:ljwp-assign}\\[1ex]
  \mathrm{ewp}_{E}[\vx <* \sexpr](g)(m)
  &=\mathbb E_{v\sim\sexpr(m)}\bigl[g(m[\vx\mapped v])\bigr],
  \nonumber\\
  \mathrm{wp}_{G}[\vx <* \sexpr](Q)(m)
  &=\forall v\in\supp(\sexpr(m)).\ Q(m[\vx\mapped v]),
  \label{eq:ljwp-sample}\\[1ex]
  \mathrm{ewp}_{E}[C;D](g)
  &=\mathrm{ewp}_{E}[C]\bigl(\mathrm{ewp}_{E}[D](g)\bigr),
  \nonumber\\
  \mathrm{wp}_{G}[C;D](Q)
  &=\mathrm{wp}_{G}[C]\bigl(\mathrm{wp}_{G}[D](Q)\bigr),
  \label{eq:ljwp-seq}\\[1ex]
  \mathrm{ewp}_{E}[\ITE{\bexpr}{C_1}{C_2}](g)(m)
  &=\begin{cases}
      \mathrm{ewp}_{E}[C_1](g)(m),&\bexpr(m),\\
      \mathrm{ewp}_{E}[C_2](g)(m),&\neg\bexpr(m),
    \end{cases}
  \nonumber\\
  \mathrm{wp}_{G}[\ITE{\bexpr}{C_1}{C_2}](Q)(m)
  &=\begin{cases}
      \mathrm{wp}_{G}[C_1](Q)(m),&\bexpr(m),\\
      \mathrm{wp}_{G}[C_2](Q)(m),&\neg\bexpr(m),
    \end{cases}
  \label{eq:ljwp-if}\\[1ex]
  \mathrm{ewp}_{E}[\mathsf{jump}\ \ell](g)&=E(\ell),
  \nonumber\\
  \mathrm{wp}_{G}[\mathsf{jump}\ \ell](Q)&=G(\ell),
  \label{eq:ljwp-jump}\\[1ex]
  \mathrm{ewp}_{E}[\ell\{C\}](g)&=\mathrm{ewp}_{E[\ell\mapsto g]}[C](g),
  \nonumber\\
  \mathrm{wp}_{G}[\ell\{C\}](Q)&=\mathrm{wp}_{G[\ell\mapsto Q]}[C](Q).
  \label{eq:ljwp-label}
\end{align}
\end{lemma}
\begin{proof}
Fix $m$.

\paragraph{Skip.}
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[\SKIP](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{r\sim\llbracket\SKIP\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-sem-skip}}{=}
  \mathbb E_{r\sim\operatorname{return}(\mathsf{Ok}(m))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)\bigl(\mathsf{Ok}(m)\bigr)\\
  &\overset{\eqref{eq:ljwp-okmap}}{=}
  g(m),
  \end{aligned}
\]
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\SKIP](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall r\in\supp(\llbracket\SKIP\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-sem-skip}}{\iff}
  \forall r\in\supp\bigl(\operatorname{return}(\mathsf{Ok}(m))\bigr).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)\bigl(\mathsf{Ok}(m)\bigr)\\
  &\overset{\eqref{eq:ljwp-okmap}}{\iff}
  Q(m).
  \end{aligned}
\]

\paragraph{Assignment.}
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[\vx <- \expression{e}](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{r\sim\llbracket \vx <- \expression{e}\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-sem-assign}}{=}
  \mathbb E_{r\sim\operatorname{return}(\mathsf{Ok}(m[\vx\mapped\expression{e}(m)]))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)\bigl(\mathsf{Ok}(m[\vx\mapped\expression{e}(m)])\bigr)\\
  &\overset{\eqref{eq:ljwp-okmap}}{=}
  g(m[\vx\mapped\expression{e}(m)]),
  \end{aligned}
\]
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\vx <- \expression{e}](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall r\in\supp(\llbracket \vx <- \expression{e}\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-sem-assign}}{\iff}
  \forall r\in\supp\bigl(\operatorname{return}(\mathsf{Ok}(m[\vx\mapped\expression{e}(m)]))\bigr).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)\bigl(\mathsf{Ok}(m[\vx\mapped\expression{e}(m)])\bigr)\\
  &\overset{\eqref{eq:ljwp-okmap}}{\iff}
  Q(m[\vx\mapped\expression{e}(m)]).
  \end{aligned}
\]

\paragraph{Sampling.}
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[\vx <* \sexpr](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{r\sim\llbracket\vx <* \sexpr\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-sem-sample}}{=}
  \mathbb E_{r\sim\operatorname{bind}(\sexpr(m),\
    \lambda v.\,\operatorname{return}(\mathsf{Ok}(m[\vx\mapped v])))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-bind}}{=}
  \mathbb E_{v\sim\sexpr(m)}\Bigl[
  \mathbb E_{r\sim\operatorname{return}(\mathsf{Ok}(m[\vx\mapped v]))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\Bigr]\\
  &\overset{\eqref{eq:lj-return}}{=}
  \mathbb E_{v\sim\sexpr(m)}\bigl[
  (E;\mathsf{Ok}\mapsto g)(\mathsf{Ok}(m[\vx\mapped v]))\bigr]\\
  &\overset{\eqref{eq:ljwp-okmap}}{=}
  \mathbb E_{v\sim\sexpr(m)}\bigl[g(m[\vx\mapped v])\bigr],
  \end{aligned}
\]
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\vx <* \sexpr](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall r\in\supp(\llbracket\vx <* \sexpr\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-sem-sample}}{\iff}
  \forall r\in\supp\bigl(\operatorname{bind}(\sexpr(m),\
    \lambda v.\,\operatorname{return}(\mathsf{Ok}(m[\vx\mapped v])))\bigr).\\
  &\qquad\qquad(G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-bind}}{\iff}
  \forall v\in\supp(\sexpr(m)).\
  \forall r\in\supp\bigl(\operatorname{return}(\mathsf{Ok}(m[\vx\mapped v]))\bigr).\\
  &\qquad\qquad(G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-return}}{\iff}
  \forall v\in\supp(\sexpr(m)).\
  (G;\mathsf{Ok}\mapsto Q)(\mathsf{Ok}(m[\vx\mapped v]))\\
  &\overset{\eqref{eq:ljwp-okmap}}{\iff}
  \forall v\in\supp(\sexpr(m)).\ Q(m[\vx\mapped v]).
  \end{aligned}
\]

\paragraph{Sequence.}
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[C;D](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{s\sim\llbracket C;D\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\overset{\eqref{eq:lj-sem-seq}}{=}
  \mathbb E_{s\sim\mathrm{dlet}_{\mathrm{RM}}\ m'\leftarrow\llbracket C\rrbracket(m)\
  \mathrm{in}\ \llbracket D\rrbracket(m')}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\overset{\eqref{eq:lj-dlet-exp}}{=}
  \mathbb E_{r\sim\llbracket C\rrbracket(m)}
  \Bigl[\Bigl(E;\ \mathsf{Ok}\mapsto\lambda m'.\
  \mathbb E_{s\sim\llbracket D\rrbracket(m')}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr)(r)\Bigr]\\
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{r\sim\llbracket C\rrbracket(m)}
  \Bigl[\bigl(E;\ \mathsf{Ok}\mapsto\mathrm{ewp}_{E}[D](g)\bigr)(r)\Bigr]\\
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathrm{ewp}_{E}[C]\bigl(\mathrm{ewp}_{E}[D](g)\bigr)(m),
  \end{aligned}
\]
where the fourth step uses \eqref{eq:ljwp-wp} for $D$ at every $m'$, so
that the $\lambda$-term is the function $\mathrm{ewp}_{E}[D](g)$, and the
last step uses \eqref{eq:ljwp-wp} for $C$ with post-expectation
$\mathrm{ewp}_{E}[D](g)$. For the guard,
\[
  \begin{aligned}
  \mathrm{wp}_{G}[C;D](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall s\in\supp(\llbracket C;D\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\overset{\eqref{eq:lj-sem-seq}}{\iff}
  \forall s\in\supp\bigl(\mathrm{dlet}_{\mathrm{RM}}\ m'\leftarrow\llbracket C\rrbracket(m)\
  \mathrm{in}\ \llbracket D\rrbracket(m')\bigr).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\overset{\eqref{eq:lj-dlet-guard}}{\iff}
  \forall r\in\supp(\llbracket C\rrbracket(m)).\
  \Bigl(G;\ \mathsf{Ok}\mapsto\lambda m'.\
  \forall s\in\supp(\llbracket D\rrbracket(m')).\
  (G;\mathsf{Ok}\mapsto Q)(s)\Bigr)(r)\\
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall r\in\supp(\llbracket C\rrbracket(m)).\
  \bigl(G;\ \mathsf{Ok}\mapsto\mathrm{wp}_{G}[D](Q)\bigr)(r)\\
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \mathrm{wp}_{G}[C]\bigl(\mathrm{wp}_{G}[D](Q)\bigr)(m).
  \end{aligned}
\]

\paragraph{Conditional.}
Suppose $\bexpr(m)$. Then
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[\ITE{\bexpr}{C_1}{C_2}](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{r\sim\llbracket\ITE{\bexpr}{C_1}{C_2}\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-sem-if}}{=}
  \mathbb E_{r\sim\llbracket C_1\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathrm{ewp}_{E}[C_1](g)(m),
  \end{aligned}
\]
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\ITE{\bexpr}{C_1}{C_2}](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall r\in\supp(\llbracket\ITE{\bexpr}{C_1}{C_2}\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-sem-if}}{\iff}
  \forall r\in\supp(\llbracket C_1\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \mathrm{wp}_{G}[C_1](Q)(m).
  \end{aligned}
\]
The case $\neg\bexpr(m)$ is the same with $C_2$ in place of $C_1$.

\paragraph{Jump.}
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[\mathsf{jump}\ \ell](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{r\sim\llbracket \mathsf{jump}\ \ell\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-sem-jump}}{=}
  \mathbb E_{r\sim\operatorname{return}(\mathsf{Jump}(\ell,m))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)\bigl(\mathsf{Jump}(\ell,m)\bigr)\\
  &\overset{\eqref{eq:ljwp-okmap}}{=}
  E(\ell)(m),
  \end{aligned}
\]
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\mathsf{jump}\ \ell](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall r\in\supp(\llbracket \mathsf{jump}\ \ell\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-sem-jump}}{\iff}
  \forall r\in\supp\bigl(\operatorname{return}(\mathsf{Jump}(\ell,m))\bigr).\
  (G;\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)\bigl(\mathsf{Jump}(\ell,m)\bigr)\\
  &\overset{\eqref{eq:ljwp-okmap}}{\iff}
  G(\ell)(m).
  \end{aligned}
\]
Neither $g$ nor $Q$ appears in the result, since $\mathsf{jump}\ \ell$
never terminates normally.

\paragraph{Label.}
Let $K$ be the function of $r$ in \eqref{eq:lj-sem-label}, so that
$\llbracket\ell\{C\}\rrbracket(m)=\operatorname{bind}(\llbracket C\rrbracket(m),K)$.
For the expectation,
\[
  \begin{aligned}
  \mathrm{ewp}_{E}[\ell\{C\}](g)(m)
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathbb E_{s\sim\llbracket \ell\{C\}\rrbracket(m)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\overset{\eqref{eq:lj-sem-label}}{=}
  \mathbb E_{s\sim\operatorname{bind}(\llbracket C\rrbracket(m),K)}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\overset{\eqref{eq:lj-bind}}{=}
  \mathbb E_{r\sim\llbracket C\rrbracket(m)}
  \Bigl[\mathbb E_{s\sim K(r)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\Bigr]\\
  &\overset{(*)}{=}
  \mathbb E_{r\sim\llbracket C\rrbracket(m)}
  \bigl[(E[\ell\mapsto g];\mathsf{Ok}\mapsto g)(r)\bigr]\\
  &\overset{\eqref{eq:ljwp-wp}}{=}
  \mathrm{ewp}_{E[\ell\mapsto g]}[C](g)(m),
  \end{aligned}
\]
where $(*)$ is the following equation, shown for every $r$:
\[
  \mathbb E_{s\sim K(r)}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]
  =(E[\ell\mapsto g];\mathsf{Ok}\mapsto g)(r).
\]
If $r=\mathsf{Ok}(m')$:
\[
  \begin{aligned}
  &\mathbb E_{s\sim K(\mathsf{Ok}(m'))}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-sem-label}}{=}
  \mathbb E_{s\sim\operatorname{return}(\mathsf{Ok}(m'))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)(\mathsf{Ok}(m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  g(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  (E[\ell\mapsto g];\mathsf{Ok}\mapsto g)(\mathsf{Ok}(m')).
  \end{aligned}
\]
If $r=\mathsf{Jump}(\ell,m')$:
\[
  \begin{aligned}
  &\mathbb E_{s\sim K(\mathsf{Jump}(\ell,m'))}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-sem-label}}{=}
  \mathbb E_{s\sim\operatorname{return}(\mathsf{Ok}(m'))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)(\mathsf{Ok}(m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  g(m')\\
  &\qquad\overset{\eqref{eq:lj-envupd}}{=}
  E[\ell\mapsto g](\ell)(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  (E[\ell\mapsto g];\mathsf{Ok}\mapsto g)(\mathsf{Jump}(\ell,m')).
  \end{aligned}
\]
If $r=\mathsf{Jump}(\ell',m')$ with $\ell'\neq\ell$:
\[
  \begin{aligned}
  &\mathbb E_{s\sim K(\mathsf{Jump}(\ell',m'))}\bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-sem-label}}{=}
  \mathbb E_{s\sim\operatorname{return}(\mathsf{Jump}(\ell',m'))}
  \bigl[(E;\mathsf{Ok}\mapsto g)(s)\bigr]\\
  &\qquad\overset{\eqref{eq:lj-return}}{=}
  (E;\mathsf{Ok}\mapsto g)(\mathsf{Jump}(\ell',m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  E(\ell')(m')\\
  &\qquad\overset{\eqref{eq:lj-envupd}}{=}
  E[\ell\mapsto g](\ell')(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{=}
  (E[\ell\mapsto g];\mathsf{Ok}\mapsto g)(\mathsf{Jump}(\ell',m')).
  \end{aligned}
\]
For the guard,
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\ell\{C\}](Q)(m)
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \forall s\in\supp(\llbracket \ell\{C\}\rrbracket(m)).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\overset{\eqref{eq:lj-sem-label}}{\iff}
  \forall s\in\supp\bigl(\operatorname{bind}(\llbracket C\rrbracket(m),K)\bigr).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\overset{\eqref{eq:lj-bind}}{\iff}
  \forall r\in\supp(\llbracket C\rrbracket(m)).\
  \forall s\in\supp(K(r)).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\overset{(**)}{\iff}
  \forall r\in\supp(\llbracket C\rrbracket(m)).\
  (G[\ell\mapsto Q];\mathsf{Ok}\mapsto Q)(r)\\
  &\overset{\eqref{eq:ljwp-wpG}}{\iff}
  \mathrm{wp}_{G[\ell\mapsto Q]}[C](Q)(m),
  \end{aligned}
\]
where $(**)$ is the following equivalence, shown for every $r$:
\[
  \forall s\in\supp(K(r)).\ (G;\mathsf{Ok}\mapsto Q)(s)
  \iff(G[\ell\mapsto Q];\mathsf{Ok}\mapsto Q)(r).
\]
If $r=\mathsf{Ok}(m')$:
\[
  \begin{aligned}
  &\forall s\in\supp(K(\mathsf{Ok}(m'))).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-sem-label}}{\iff}
  \forall s\in\supp(\operatorname{return}(\mathsf{Ok}(m'))).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)(\mathsf{Ok}(m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  Q(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  (G[\ell\mapsto Q];\mathsf{Ok}\mapsto Q)(\mathsf{Ok}(m')).
  \end{aligned}
\]
If $r=\mathsf{Jump}(\ell,m')$:
\[
  \begin{aligned}
  &\forall s\in\supp(K(\mathsf{Jump}(\ell,m'))).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-sem-label}}{\iff}
  \forall s\in\supp(\operatorname{return}(\mathsf{Ok}(m'))).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)(\mathsf{Ok}(m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  Q(m')\\
  &\qquad\overset{\eqref{eq:lj-envupd}}{\iff}
  G[\ell\mapsto Q](\ell)(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  (G[\ell\mapsto Q];\mathsf{Ok}\mapsto Q)(\mathsf{Jump}(\ell,m')).
  \end{aligned}
\]
If $r=\mathsf{Jump}(\ell',m')$ with $\ell'\neq\ell$:
\[
  \begin{aligned}
  &\forall s\in\supp(K(\mathsf{Jump}(\ell',m'))).\ (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-sem-label}}{\iff}
  \forall s\in\supp(\operatorname{return}(\mathsf{Jump}(\ell',m'))).\
  (G;\mathsf{Ok}\mapsto Q)(s)\\
  &\qquad\overset{\eqref{eq:lj-return}}{\iff}
  (G;\mathsf{Ok}\mapsto Q)(\mathsf{Jump}(\ell',m'))\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  G(\ell')(m')\\
  &\qquad\overset{\eqref{eq:lj-envupd}}{\iff}
  G[\ell\mapsto Q](\ell')(m')\\
  &\qquad\overset{\eqref{eq:ljwp-okmap}}{\iff}
  (G[\ell\mapsto Q];\mathsf{Ok}\mapsto Q)(\mathsf{Jump}(\ell',m')).
  \end{aligned}
\]
\end{proof}

\subsection{Rules and Their Soundness}\label{app:lj-rules}

For each rule we show that its conclusion is valid in the sense of
\eqref{eq:ljwp-validity} whenever its premises are. Each proof unfolds
\eqref{eq:ljwp-validity} into a guard condition and a numerical condition
and discharges both with \Cref{lem:ljwp-equations}. We use that
$\bowtie\in\{\leq,=,\geq\}$ is reflexive.

\subsubsection{Skip}\label{app:lj-rule-skip}
\[
  \frac{}{
    G\mid E\vdash\{Q\mid g\}_{\bowtie}\ \SKIP\ \{Q\mid g\}
  }
  \;\textsc{Skip}
\]

\begin{proof}
By \eqref{eq:ljwp-validity}, we must show that for every memory $m$,
\[
  Q(m)\ \Longrightarrow\
  \mathrm{wp}_{G}[\SKIP](Q)(m)
  \ \land\
  g(m)\mathrel{\bowtie}\mathrm{ewp}_{E}[\SKIP](g)(m).
\]
So let $m$ be a memory and assume $Q(m)$.
For the guard condition, by \eqref{eq:ljwp-skip},
\[
  \mathrm{wp}_{G}[\SKIP](Q)(m)=Q(m),
\]
which is exactly the assumption.
For the numerical condition, by \eqref{eq:ljwp-skip},
\[
  \mathrm{ewp}_{E}[\SKIP](g)(m)=g(m),
\]
so we must show $g(m)\mathrel{\bowtie}g(m)$, which holds by reflexivity
of $\bowtie$.
\end{proof}

\subsubsection{Assignment}\label{app:lj-rule-assign}
For a predicate $Q$ and an expectation $g$ write
$Q[\vx\mapped\expression{e}](m)\defsym Q(m[\vx\mapped\expression{e}(m)])$ and
$g[\vx\mapped\expression{e}](m)\defsym g(m[\vx\mapped\expression{e}(m)])$.
\[
  \frac{}{
    G\mid E\vdash
    \{Q[\vx\mapped\expression{e}]\mid g[\vx\mapped\expression{e}]\}_{\bowtie}\
    \vx <- \expression{e}\
    \{Q\mid g\}
  }
  \;\textsc{Assign}
\]

\begin{proof}
Fix $m$ with $Q[\vx\mapped\expression{e}](m)$, that is,
$Q(m[\vx\mapped\expression{e}(m)])$.
For the guard condition,
\[
  \mathrm{wp}_{G}[\vx <- \expression{e}](Q)(m)
  \overset{\eqref{eq:ljwp-assign}}{=}
  Q(m[\vx\mapped\expression{e}(m)]),
\]
which holds by assumption.
For the numerical condition,
\[
  \mathrm{ewp}_{E}[\vx <- \expression{e}](g)(m)
  \overset{\eqref{eq:ljwp-assign}}{=}
  g(m[\vx\mapped\expression{e}(m)])
  =
  g[\vx\mapped\expression{e}](m),
\]
so we must show
$g[\vx\mapped\expression{e}](m)\mathrel{\bowtie}g[\vx\mapped\expression{e}](m)$,
which holds by reflexivity of $\bowtie$.
\end{proof}

\subsubsection{Sampling}\label{app:lj-rule-sample}
For a predicate $Q$ and an expectation $g$ write
$\bigl(\forall v\in\supp(\sexpr).\ Q[\vx\mapped v]\bigr)(m)\defsym
\forall v\in\supp(\sexpr(m)).\ Q(m[\vx\mapped v])$ and
$\mathbb E_{v\sim\sexpr}\bigl[g[\vx\mapped v]\bigr](m)\defsym
\mathbb E_{v\sim\sexpr(m)}\bigl[g(m[\vx\mapped v])\bigr]$.
\[
  \frac{}{
    G\mid E\vdash
    \bigl\{\forall v\in\supp(\sexpr).\ Q[\vx\mapped v]\bigm|
    \mathbb E_{v\sim\sexpr}\bigl[g[\vx\mapped v]\bigr]\bigr\}_{\bowtie}\
    \vx <* \sexpr\
    \{Q\mid g\}
  }
  \;\textsc{Sample}
\]

\begin{proof}
Fix $m$ with $\forall v\in\supp(\sexpr(m)).\ Q(m[\vx\mapped v])$.
For the guard condition,
\[
  \mathrm{wp}_{G}[\vx <* \sexpr](Q)(m)
  \overset{\eqref{eq:ljwp-sample}}{=}
  \forall v\in\supp(\sexpr(m)).\ Q(m[\vx\mapped v]),
\]
which holds by assumption.
For the numerical condition,
\[
  \mathrm{ewp}_{E}[\vx <* \sexpr](g)(m)
  \overset{\eqref{eq:ljwp-sample}}{=}
  \mathbb E_{v\sim\sexpr(m)}\bigl[g(m[\vx\mapped v])\bigr]
  =
  \mathbb E_{v\sim\sexpr}\bigl[g[\vx\mapped v]\bigr](m),
\]
so the pre-expectation equals $\mathrm{ewp}_{E}[\vx <* \sexpr](g)(m)$
and the condition holds by reflexivity of $\bowtie$.
\end{proof}

\subsubsection{Sequence}\label{app:lj-rule-seq}
The intermediate assertion $\{R\mid h\}$ connects the two premises: $R$
holds after every normal exit of $C$, and $h$ is the value carried from
$C$ to $D$ at that point. Jumps out of $C$ skip $D$ and are handled by the
shared environment $G\mid E$.
\[
  \frac{
    G\mid E\vdash\{P\mid f\}_{\bowtie}\ C\ \{R\mid h\}
    \qquad
    G\mid E\vdash\{R\mid h\}_{\bowtie}\ D\ \{Q\mid g\}
  }{
    G\mid E\vdash\{P\mid f\}_{\bowtie}\ C;D\ \{Q\mid g\}
  }
  \;\textsc{Seq}
\]

\begin{proof}
Fix $m$ with $P(m)$. The first premise gives
\[
  \text{(i)}\ \ \mathrm{wp}_{G}[C](R)(m),
  \qquad
  \text{(ii)}\ \ f(m)\mathrel{\bowtie}\mathrm{ewp}_{E}[C](h)(m),
\]
and the second premise gives, for every $m'$ with $R(m')$,
\[
  \text{(iii)}\ \ \mathrm{wp}_{G}[D](Q)(m'),
  \qquad
  \text{(iv)}\ \ h(m')\mathrel{\bowtie}\mathrm{ewp}_{E}[D](g)(m').
\]
By \eqref{eq:ljwp-wpG}, (i) says that $(G;\mathsf{Ok}\mapsto R)(r)$ holds
for every $r\in\supp(\llbracket C\rrbracket(m))$. Fix such an $r$.
\begin{itemize}
  \item If $r=\mathsf{Ok}(m')$, then $R(m')$ by \eqref{eq:ljwp-okmap}, so
    (iii) and (iv) apply at $m'$ and give, by \eqref{eq:ljwp-okmap},
    \[
      \bigl(G;\mathsf{Ok}\mapsto\mathrm{wp}_{G}[D](Q)\bigr)(r)
      \quad\text{and}\quad
      (E;\mathsf{Ok}\mapsto h)(r)\mathrel{\bowtie}
      \bigl(E;\mathsf{Ok}\mapsto\mathrm{ewp}_{E}[D](g)\bigr)(r).
    \]
  \item If $r=\mathsf{Jump}(\ell,m')$, then by \eqref{eq:ljwp-okmap} the
    guard $\bigl(G;\mathsf{Ok}\mapsto\mathrm{wp}_{G}[D](Q)\bigr)(r)$ is
    $G(\ell)(m')$, which holds by (i), and both
    $(E;\mathsf{Ok}\mapsto h)(r)$ and
    $\bigl(E;\mathsf{Ok}\mapsto\mathrm{ewp}_{E}[D](g)\bigr)(r)$ are
    $E(\ell)(m')$, so they are related by $\bowtie$ by reflexivity.
\end{itemize}
For the guard condition, the first part of both cases gives, for every
$r\in\supp(\llbracket C\rrbracket(m))$,
$\bigl(G;\mathsf{Ok}\mapsto\mathrm{wp}_{G}[D](Q)\bigr)(r)$, hence
\[
  \mathrm{wp}_{G}[C]\bigl(\mathrm{wp}_{G}[D](Q)\bigr)(m)
  \overset{\eqref{eq:ljwp-seq}}{=}
  \mathrm{wp}_{G}[C;D](Q)(m)
\]
by \eqref{eq:ljwp-wpG}.
For the numerical condition, the second part of both cases relates the
two brackets by $\bowtie$ at every $r$ in the support. An expectation only
depends on the values on the support and is monotone, so averaging over
$r\sim\llbracket C\rrbracket(m)$ keeps $\bowtie$, and by
\eqref{eq:ljwp-wp}
\[
  f(m)
  \overset{\text{(ii)}}{\mathrel{\bowtie}}
  \mathrm{ewp}_{E}[C](h)(m)
  \mathrel{\bowtie}
  \mathrm{ewp}_{E}[C]\bigl(\mathrm{ewp}_{E}[D](g)\bigr)(m)
  \overset{\eqref{eq:ljwp-seq}}{=}
  \mathrm{ewp}_{E}[C;D](g)(m).
\]
Since $\bowtie$ is transitive,
$f(m)\mathrel{\bowtie}\mathrm{ewp}_{E}[C;D](g)(m)$.
\end{proof}

\subsubsection{Conditional}\label{app:lj-rule-if}
\[
  \frac{
    G\mid E\vdash\{\bexpr\land P\mid f\}_{\bowtie}\ C_1\ \{Q\mid g\}
    \qquad
    G\mid E\vdash\{\neg\bexpr\land P\mid f\}_{\bowtie}\ C_2\ \{Q\mid g\}
  }{
    G\mid E\vdash\{P\mid f\}_{\bowtie}\ \ITE{\bexpr}{C_1}{C_2}\ \{Q\mid g\}
  }
  \;\textsc{If}
\]

\begin{proof}
Fix $m$ with $P(m)$, and suppose $\bexpr(m)$. Then
$(\bexpr\land P)(m)$, so the first premise gives
$\mathrm{wp}_{G}[C_1](Q)(m)$ and
$f(m)\mathrel{\bowtie}\mathrm{ewp}_{E}[C_1](g)(m)$. By
\eqref{eq:ljwp-if}, since $\bexpr(m)$,
\[
  \begin{aligned}
  \mathrm{wp}_{G}[\ITE{\bexpr}{C_1}{C_2}](Q)(m)
  &=\mathrm{wp}_{G}[C_1](Q)(m),\\
  \mathrm{ewp}_{E}[\ITE{\bexpr}{C_1}{C_2}](g)(m)
  &=\mathrm{ewp}_{E}[C_1](g)(m),
  \end{aligned}
\]
so both conditions hold. The case $\neg\bexpr(m)$ is the same, using the
second premise and $C_2$.
\end{proof}

\subsubsection{Jump}\label{app:lj-rule-jump}
A jump never ends normally, so the postcondition $\{Q\mid g\}$ plays no
role.
\[
  \frac{
    \forall m.\ P(m)\Longrightarrow G(\ell)(m)
    \qquad
    \forall m.\ P(m)\Longrightarrow
      f(m)\mathrel{\bowtie}E(\ell)(m)
  }{
    G\mid E\vdash
    \{P\mid f\}_{\bowtie}\ \mathsf{jump}\ \ell\ \{Q\mid g\}
  }
  \;\textsc{Jump}
\]

\begin{proof}
Fix $m$ with $P(m)$.
For the guard condition,
\[
  \mathrm{wp}_{G}[\mathsf{jump}\ \ell](Q)(m)
  \overset{\eqref{eq:ljwp-jump}}{=}
  G(\ell)(m),
\]
which holds by the first premise.
For the numerical condition,
\[
  \mathrm{ewp}_{E}[\mathsf{jump}\ \ell](g)(m)
  \overset{\eqref{eq:ljwp-jump}}{=}
  E(\ell)(m),
\]
and $f(m)\mathrel{\bowtie}E(\ell)(m)$ holds by the second premise.
\end{proof}

\subsubsection{Label}\label{app:lj-rule-label}
Inside $\ell\{C\}$, a jump to $\ell$ ends the block normally. So in the
premise the label $\ell$ is given the postcondition itself: guard $Q$ and
weight $g$. Jumps to other labels keep their guard and weight from
$G\mid E$.
\[
  \frac{
    G[\ell\mapsto Q]\mid E[\ell\mapsto g]\vdash
    \{P\mid f\}_{\bowtie}\ C\ \{Q\mid g\}
  }{
    G\mid E\vdash
    \{P\mid f\}_{\bowtie}\ \ell\{C\}\ \{Q\mid g\}
  }
  \;\textsc{Label}
\]

\begin{proof}
Fix $m$ with $P(m)$. The premise gives
\[
  \mathrm{wp}_{G[\ell\mapsto Q]}[C](Q)(m)
  \qquad\text{and}\qquad
  f(m)\mathrel{\bowtie}\mathrm{ewp}_{E[\ell\mapsto g]}[C](g)(m).
\]
By \eqref{eq:ljwp-label},
\[
  \begin{aligned}
  \mathrm{wp}_{G[\ell\mapsto Q]}[C](Q)(m)
  &=\mathrm{wp}_{G}[\ell\{C\}](Q)(m),\\
  \mathrm{ewp}_{E[\ell\mapsto g]}[C](g)(m)
  &=\mathrm{ewp}_{E}[\ell\{C\}](g)(m),
  \end{aligned}
\]
so both conditions hold.
\end{proof}

\section*{Canonical Witnesses: Worked Examples}\label{app:canonical-examples}

\paragraph{The canonical witness sequence.}
Given a loop $\WHILE\bexpr\DO\cmd$ with invariant $P$, postcondition expectation $g$, and
letting $q_n(m)=\Pr(T_m > n)$ denote the tail probabilities of the loop's
hitting time $T_m$ starting from state $m$, the \emph{canonical} witness sequence is
\begin{equation}\label{eq:canonical-r}
  r_n(m,c) \;\defsym\; (c + n + 1)\,q_n(m) \;+\; \sum_{j=n+1}^{\infty} q_j(m).
\end{equation}
This can equivalently be written as $r_n(m,c) = c\cdot q_n(m) + \mathbb{E}[T_m \cdot [T_m > n]]$,
which makes plain why it decreases to zero (the truncated expectation vanishes
as $n\to\infty$) and why it starts at $g$ (the Abel/tail-sum identity
$\sum_{n\geq 0} q_n(m) = \mathbb{E}[T_m]$ gives $r_0(m,c)=c+\mathbb{E}[T_m]=g(m,c)$).

We verify the four premises \textbf{P1}--\textbf{P4} of the \textsc{While-lb} rule on two concrete programs.

\subsection*{Example 1: Geometric Termination}

\paragraph{Program.}
\[
  C_{\text{geo}} \;\defsym\; \WHILE{b}{\;b \mathrel{:=} \mathrm{Ber}(p);\; c \mathrel{:=} c+1\;}
\]
The loop body samples a fresh Bernoulli($p$) bit and increments the counter $c$.
The loop terminates when $b=\mathbf{false}$, which happens geometrically:
$T_{b=\mathbf{true}} \sim \mathrm{Geom}(1-p)$, so $\mathbb{E}[T]=\tfrac{1}{1-p}$.

\paragraph{Postcondition and invariant.}
Take $P \equiv \mathbf{true}$ and
\[
  g(b,c) \;\defsym\; c + \frac{[b]}{1-p}.
\]
At the false-branch terminal state, $g(\mathbf{false},c)=c$, which records the
final value of $c$ (number of loop iterations taken).

\paragraph{Tail probabilities.}
$q_n(\mathbf{true}) = p^n$ and $q_n(\mathbf{false}) = 0$.

\paragraph{Canonical witnesses.}
\begin{align*}
  r_n(\mathbf{true},c) &= (c+n+1)\,p^n + \frac{p^{n+1}}{1-p}, \\
  r_n(\mathbf{false},c) &= 0.
\end{align*}

\paragraph{Verification of P1.}
We need $\vdash_{\leq}\HT{b\land P}{g}{C_{\text{body}}}{P}{g}$, i.e.\
$\mathrm{wp}[b\mathrel{:=}\mathrm{Ber}(p);\,c\mathrel{:=}c+1](g) \geq g$.
\begin{align*}
  \mathrm{wp}[C_{\text{body}}](g)(b,c)
    &= p\cdot g(\mathbf{true},c+1) + (1-p)\cdot g(\mathbf{false},c+1) \\
    &= p\!\left(c+1+\tfrac{1}{1-p}\right) + (1-p)(c+1) \\
    &= c + 1 + \tfrac{p}{1-p}
     = c + \tfrac{1}{1-p}
     = g(\mathbf{true},c).
\end{align*}
Equality holds, so P1 is satisfied.

\paragraph{Verification of P2.}
We need $r_0 \geq g$ on $B\land P$, i.e.\ at $b=\mathbf{true}$:
\[
  r_0(\mathbf{true},c)
  = (c+0+1)\,p^0 + \frac{p^1}{1-p}
  = c + 1 + \frac{p}{1-p}
  = c + \frac{1}{1-p}
  = g(\mathbf{true},c).
\]

\paragraph{Verification of P3.}
We need $\mathrm{wp}[C_{\text{body}}](r_n) \leq r_{n+1}$. Compute at $b=\mathbf{true}$:
\begin{align*}
  \mathrm{wp}[C_{\text{body}}](r_n)(\mathbf{true},c)
    &= p\cdot r_n(\mathbf{true},c+1) + (1-p)\cdot r_n(\mathbf{false},c+1) \\
    &= p\!\left[(c+n+2)\,p^n + \frac{p^{n+1}}{1-p}\right] + 0 \\
    &= (c+n+2)\,p^{n+1} + \frac{p^{n+2}}{1-p}.
\end{align*}
Meanwhile,
\[
  r_{n+1}(\mathbf{true},c)
  = (c+n+2)\,p^{n+1} + \frac{p^{n+2}}{1-p}.
\]
These are equal, so P3 holds with equality.

\paragraph{Verification of P4.}
On $B\land P$ (i.e.\ $b=\mathbf{true}$):
\[
  \lim_{n\to\infty} r_n(\mathbf{true},c)
  = \lim_{n\to\infty}\!\left[(c+n+1)\,p^n + \frac{p^{n+1}}{1-p}\right] = 0,
\]
since $p\in(0,1)$ so exponential decay dominates the polynomial prefactor.

\subsection*{Example 2: Deterministic Countdown}

\paragraph{Program.}
\[
  C_{\text{det}} \;\defsym\; \WHILE{x>0}{\;x \mathrel{:=} x-1;\; c \mathrel{:=} c+1\;}
\]
The loop body decrements $x$ and increments $c$ deterministically.
Starting from $x=k\geq 0$, the loop runs exactly $k$ iterations: $T_k = k$.

\paragraph{Postcondition and invariant.}
Take $P \equiv \mathbf{true}$ and
\[
  g(x,c) \;\defsym\; c + x.
\]

\paragraph{Tail probabilities.}
Because $T_x = x$ deterministically,
$q_n(x) = \Pr(T_x > n) = [x > n]$.

\paragraph{Canonical witnesses.}
\begin{equation*}
  r_n(x,c) = (c+n+1)\,[x>n] + \sum_{j=n+1}^{\infty}[x>j]
            = (c+n+1)\,[x>n] + \max(x-n-1,\,0).
\end{equation*}
For $x > n$ this simplifies to $(c+n+1) + (x-n-1) = c+x$.
For $x \leq n$ both terms vanish, giving $r_n(x,c) = 0$.
Hence
\[
  r_n(x,c) = (c + x)\cdot[x > n].
\]

\paragraph{Verification of P1.}
\begin{align*}
  \mathrm{wp}[C_{\text{body}}](g)(x,c)
    &= g(x-1,\,c+1) = (c+1)+(x-1) = c+x = g(x,c).
\end{align*}

\paragraph{Verification of P2.}
At $x>0$ (i.e.\ $B\land P$):
\[
  r_0(x,c) = (c+x)\cdot[x>0] = c+x = g(x,c).
\]

\paragraph{Verification of P3.}
\begin{align*}
  \mathrm{wp}[C_{\text{body}}](r_n)(x,c)
    &= r_n(x-1,\,c+1) \\
    &= (c+1+(x-1))\cdot[x-1>n] \\
    &= (c+x)\cdot[x>n+1] \\
    &= r_{n+1}(x,c).
\end{align*}

\paragraph{Verification of P4.}
For any fixed $x$, $[x>n]=0$ for all $n\geq x$, so
$\lim_{n\to\infty} r_n(x,c) = 0$.
