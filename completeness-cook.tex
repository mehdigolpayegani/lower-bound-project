This section proves completeness for guarded judgments with arbitrary
qualitative pre- and postconditions.  Let
$P,Q:\Mem\to\Bool$ and $f,g:\Mem\to[0,\infty]$.  For every command $C$ and
post-assertion $(Q\mid g)$, structural induction on $C$ derives the judgment
whose two precomponents are semantically exact: the qualitative component is
$\WP{C}{Q}$ and the quantitative component is $\WP{C}{g}$.  A final use of
\textsc{Consequence} then derives every semantically valid judgment
$\HT{P}{f}{C}{Q}{g}$.

Throughout,
\[
  \bowtie\ \in\ \{\,\leq,=,\geq\,\}
\]
denotes the lower, equality, or upper judgment relation. The induction is
carried out once, for an arbitrary $\bowtie$.

\subsubsection{Exactly what is proved}

For every command $C$, predicates $P,Q$, and expectations $f,g$, we prove
\begin{equation}
  \valid\HT{P}{f}{C}{Q}{g}
  \quad\Longrightarrow\quad
  \vdash\HT{P}{f}{C}{Q}{g}.
  \label{eq:cook-target}
\end{equation}
By the typed validity definition \eqref{eq:cook-validity-typed}, the premise of
\eqref{eq:cook-target} means
\begin{equation}
  \forall m\in\Mem.\quad
  P(m)\Longrightarrow
  \bigl(\WP{C}{Q}(m)\land
        f(m)\bowtie\WP{C}{g}(m)\bigr),
  \label{eq:cook-general-validity}
\end{equation}
or, equivalently,
\begin{equation}
  P\models \WP{C}{Q}\land f\bowtie\WP{C}{g}.
  \label{eq:cook-general-validity-entailment}
\end{equation}
The proof proceeds through the following exact-pre judgment:
\begin{equation}
  \vdash
  \HT{\WP{C}{Q}}{\WP{C}{g}}{C}{Q}{g}
  \label{eq:cook-general-exact-pre-target}
\end{equation}
Here $\WP{C}{Q}$ is the exact qualitative precondition for $Q$, and
$\WP{C}{g}$ is the exact quantitative pre-expectation of $g$.  Once
\eqref{eq:cook-general-exact-pre-target} has been derived, the desired judgment
follows by \textsc{Consequence}:
\begin{equation}
\frac{
  \begin{gathered}
    P\models\WP{C}{Q}\land f\bowtie\WP{C}{g}
    \\[1mm]
    \vdash\HT{\WP{C}{Q}}{\WP{C}{g}}{C}{Q}{g}
    \\[1mm]
    Q\models Q\land g\bowtie g
  \end{gathered}
}{
  \vdash\HT{P}{f}{C}{Q}{g}
}\;\textsc{Consequence}.
\label{eq:cook-general-final-consequence-outline}
\end{equation}
The first premise is \eqref{eq:cook-general-validity-entailment}, the second is
the exact-pre judgment, and the third holds by reflexivity.  It therefore
suffices to prove \eqref{eq:cook-general-exact-pre-target} by structural
induction on $C$.

\subsubsection{The rules and semantic equations used}

The \textsc{Skip}, \textsc{Assign}, \textsc{Sample}, \textsc{Seq}, and
\textsc{If} rules are those in \Cref{fig:hoare:l-u}.  They are parameterised
by $\bowtie$ and therefore cover lower, equality, and upper judgments at once.

\paragraph{The guarded consequence rule.}
Following the actual guarded consequence rule and the notation in the rest of
this note, we use
\begin{equation}
\frac{
  P\models P'\land f\bowtie f'
  \qquad
  \vdash\HT{P'}{f'}{C}{Q'}{g'}
  \qquad
  Q'\models Q\land g'\bowtie g
}{
  \vdash\HT{P}{f}{C}{Q}{g}
}\;\textsc{Consequence}.
\label{eq:cook-consequence}
\end{equation}
For $\bowtie={\leq}$ this is the lower rule, for $\bowtie={=}$ the equality
rule, and for $\bowtie={\geq}$ the upper rule.  The first comparison is
required on the new pre-guard $P$, and the last comparison is required on the
source post-guard $Q'$.  Every use of \textsc{Consequence} below is displayed
in this three-premise form with all entries already instantiated.

\paragraph{Semantic equations.}
The expectation semantics gives
\begin{align}
  \WP{\SKIP}{g} &= g,
  \label{eq:cook-wp-skip}\\
  \WP{\vx <- \expr}{g} &= g[\vx\mapped\expr],
  \label{eq:cook-wp-assign}\\
  \WP{\vx <* \sexpr}{g}
    &= \E{\sexpr}{\lambda\val.\,g[\vx\mapped\val]},
  \label{eq:cook-wp-sample}\\
  \WP{C_1;C_2}{g}
    &= \WP{C_1}{\WP{C_2}{g}},
  \label{eq:cook-wp-seq}\\
  \WP{\ITE{B}{C_1}{C_2}}{g}
    &= [B]\cdot\WP{C_1}{g}
       +[\neg B]\cdot\WP{C_2}{g}.
  \label{eq:cook-wp-if}
\end{align}
These equations follow from the denotational clauses above rather than being
additional proof rules: the first four follow from the return and bind laws,
and the conditional equation follows by splitting on $B(m)$.

For loops we use \Cref{lem:In-recursion}: for $L=\WHILE B\DO D$ and the
approximations $I_n$ of \eqref{eq:cook-loop-approx},
\[
  I_0=[\neg B]\cdot g,
  \qquad
  I_{n+1}=[B]\cdot\WP{D}{I_n}+[\neg B]\cdot g,
  \qquad
  \WP{L}{g}=\sup_n I_n.
\]

We assume that the exact preconditions, pre-expectations, and loop residuals
used below are expressible, and that the background assertion theory proves
the required semantic equalities and inequalities.  In the loop case, the
branches $\bowtie\in\{\leq,=\}$ additionally assume that the exact
pre-expectation $\WP{L}{g}$ of the loop is finite-valued, so that the
residuals $r_n$ are well defined and never form $\infty-\infty$; the branch
$\bowtie={\geq}$ needs no such assumption.

\subsubsection{The exact-pre lemma}

\begin{lemma}[Exact-pre judgments]
\label[lemma]{lem:exact-pre-derivable}
For every command $C$, every postcondition $Q$, every post-expectation $g$ in
the expressive, finitary fragment, and each
$\bowtie\in\{\leq,=,\geq\}$,
\begin{equation}
  \vdash\HT{\WP{C}{Q}}{\WP{C}{g}}{C}{Q}{g}.
  \label{eq:cook-exact-bowtie}
\end{equation}
\end{lemma}

\begin{proof}
The proof is by structural induction on $C$, with $Q$, $g$, and $\bowtie$
universally quantified.

\paragraph{Case: skip.}

The typed weakest-precondition equations are
\[
  \WP{\SKIP}{Q}=Q,
  \qquad
  \WP{\SKIP}{g}=g.
\]
Hence \textsc{Skip} gives
\[
\frac{}{\vdash\HT{Q}{g}{\SKIP}{Q}{g}}\;\textsc{Skip},
\]
which is \eqref{eq:cook-exact-bowtie} after rewriting both precomponents.

\paragraph{Case: assignment.}

The typed assignment equations are
\[
  \WP{\vx <- \expr}{Q}=Q[\vx\mapped\expr],
  \qquad
  \WP{\vx <- \expr}{g}=g[\vx\mapped\expr].
\]
The required instance of \textsc{Assign} is
\[
\frac{}{
  \vdash\HT{Q[\vx\mapped\expr]}{g[\vx\mapped\expr]}
    {\vx <- \expr}{Q}{g}
}\;\textsc{Assign}.
\]
Rewriting its two precomponents proves \eqref{eq:cook-exact-bowtie}.

\paragraph{Case: sampling.}

The typed sampling equations are
\[
  \WP{\vx <* \sexpr}{Q}
  =\forall\val\in\supp(\sexpr).\ Q[\vx\mapped\val],
  \qquad
  \WP{\vx <* \sexpr}{g}
  =\E{\sexpr}{\lambda\val.\,g[\vx\mapped\val]}.
\]
Thus \textsc{Sample} gives
\[
\frac{}{
  \vdash\HT{\forall\val\in\supp(\sexpr).\ Q[\vx\mapped\val]}
    {\E{\sexpr}{\lambda\val.\,g[\vx\mapped\val]}}
    {\vx <* \sexpr}{Q}{g}
}\;\textsc{Sample}.
\]
Rewriting its two precomponents proves \eqref{eq:cook-exact-bowtie}.

\paragraph{Case: sequential composition.}

Define
\[
  R\defsym\WP{C_2}{Q},
  \qquad
  h\defsym\WP{C_2}{g}.
\]
The two induction hypotheses are
\[
  \vdash\HT{R}{h}{C_2}{Q}{g},
  \qquad
  \vdash\HT{\WP{C_1}{R}}{\WP{C_1}{h}}{C_1}{R}{h}.
\]
They are exactly the two premises of \textsc{Seq}:
\begin{equation}
\frac{
  \vdash\HT{\WP{C_1}{R}}{\WP{C_1}{h}}{C_1}{R}{h}
  \qquad
  \vdash\HT{R}{h}{C_2}{Q}{g}
}{
  \vdash\HT{\WP{C_1}{R}}{\WP{C_1}{h}}{C_1;C_2}{Q}{g}
}\;\textsc{Seq}.
\label{eq:cook-seq-derivation}
\end{equation}
The composition equations
\[
  \WP{C_1;C_2}{Q}=\WP{C_1}{R},
  \qquad
  \WP{C_1;C_2}{g}=\WP{C_1}{h}
\]
rewrite the conclusion to \eqref{eq:cook-exact-bowtie}.

\paragraph{Case: $C$ is a conditional on $B$.}

Here $C=\ITE{B}{C_1}{C_2}$, and the structurally smaller commands are
$C_1$ and $C_2$.  Fix $Q$ and $g$.  Unfolding the semantics of the
conditional gives
\begin{align}
  B&\models \WP{C}{Q}=\WP{C_1}{Q}\ \land\ \WP{C}{g}=\WP{C_1}{g},
  \label{eq:cook-if-true}\\
  \neg B&\models \WP{C}{Q}=\WP{C_2}{Q}\ \land\ \WP{C}{g}=\WP{C_2}{g}.
  \label{eq:cook-if-false}
\end{align}
These are semantic facts, not induction hypotheses or syntactic judgments.

The two induction hypotheses, instantiated at the postcondition $Q$ and
the post-expectation $g$, are
\begin{align}
  &\vdash\HT{\WP{C_1}{Q}}{\WP{C_1}{g}}{C_1}{Q}{g},
  \label{eq:cook-if-IH-one}\\
  &\vdash\HT{\WP{C_2}{Q}}{\WP{C_2}{g}}{C_2}{Q}{g}.
  \label{eq:cook-if-IH-two}
\end{align}
They cannot be used directly as the premises of \textsc{If}, which requires
the pre-guards $B\land \WP{C}{Q}$ and $\neg B\land \WP{C}{Q}$ and the \emph{same}
pre-expectation $\WP{C}{g}$ in both branches.  \textsc{Consequence} bridges this
gap.

\paragraph{The true branch by \textsc{Consequence}.}
Substitute
\[
  P=B\land \WP{C}{Q},\quad f=\WP{C}{g},\quad P'=\WP{C_1}{Q},\quad f'=\WP{C_1}{g},
  \quad Q'=Q,\quad g'=g
\]
in \eqref{eq:cook-consequence}.  The complete rule instance is
\begin{equation}
\frac{
  \begin{array}{c}
    B\land \WP{C}{Q}\models\WP{C_1}{Q}\land \WP{C}{g}\bowtie\WP{C_1}{g}
    \\[1mm]
    \qquad
    \vdash\HT{\WP{C_1}{Q}}{\WP{C_1}{g}}{C_1}{Q}{g}
    \\[1mm]
    Q\models Q\land g\bowtie g
  \end{array}
}{
  \vdash\HT{B\land \WP{C}{Q}}{\WP{C}{g}}{C_1}{Q}{g}
}\;\textsc{Consequence}.
\label{eq:cook-if-conseq-true}
\end{equation}
The middle premise is IH~\eqref{eq:cook-if-IH-one}.  The first premise
follows from \eqref{eq:cook-if-true}: under $B$, $\WP{C}{Q}$ is
$\WP{C_1}{Q}$, and the equality $\WP{C}{g}=\WP{C_1}{g}$ implies
$\WP{C}{g}\bowtie\WP{C_1}{g}$ for every $\bowtie$.  The last premise holds
by reflexivity.

\paragraph{The false branch by \textsc{Consequence}.}
Now substitute
\[
  P=\neg B\land \WP{C}{Q},\quad f=\WP{C}{g},\quad P'=\WP{C_2}{Q},\quad f'=\WP{C_2}{g},
  \quad Q'=Q,\quad g'=g.
\]
The complete rule instance is
\begin{equation}
\frac{
  \begin{array}{c}
    \neg B\land \WP{C}{Q}\models\WP{C_2}{Q}\land \WP{C}{g}\bowtie\WP{C_2}{g}
    \\[1mm]
    \qquad
    \vdash\HT{\WP{C_2}{Q}}{\WP{C_2}{g}}{C_2}{Q}{g}
    \\[1mm]
    Q\models Q\land g\bowtie g
  \end{array}
}{
  \vdash\HT{\neg B\land \WP{C}{Q}}{\WP{C}{g}}{C_2}{Q}{g}
}\;\textsc{Consequence}.
\label{eq:cook-if-conseq-false}
\end{equation}
The middle premise is IH~\eqref{eq:cook-if-IH-two}; the first follows from
\eqref{eq:cook-if-false} in the same way; and the last is reflexive.

\paragraph{Application of \textsc{If}.}
The conclusions of \eqref{eq:cook-if-conseq-true} and
\eqref{eq:cook-if-conseq-false} are exactly the two branch premises of
\textsc{If} with pre-guard $\WP{C}{Q}$ and pre-expectation $\WP{C}{g}$:
\begin{equation}
\frac{
  \vdash\HT{B\land \WP{C}{Q}}{\WP{C}{g}}{C_1}{Q}{g}
  \qquad
  \vdash\HT{\neg B\land \WP{C}{Q}}{\WP{C}{g}}{C_2}{Q}{g}
}{
  \vdash\HT{\WP{C}{Q}}{\WP{C}{g}}{\ITE{B}{C_1}{C_2}}{Q}{g}
}\;\textsc{If}.
\label{eq:cook-if-derivation}
\end{equation}
The conclusion is \eqref{eq:cook-exact-bowtie} for the conditional.  This proves the
conditional case simultaneously for $\bowtie\in\{\leq,=,\geq\}$.

\paragraph{Case: $C$ is a loop $L$ with guard $B$ and body $D$.}
Write $L=\WHILE B\DO D$. Set
\[
  P\defsym\WP{L}{Q},
  \qquad
  V\defsym\WP{L}{g}.
\]
$P$ is the qualitative and $V$ the quantitative loop invariant.  Let $I_n$
be the approximations \eqref{eq:cook-loop-approx} for the guard $B$, the body
$D$ and the expectation $g$.  The goal \eqref{eq:cook-exact-bowtie} for $L$
reads
\[
  \vdash\HT{P}{V}{L}{Q}{g}.
\]
We derive it by the \textsc{While} rule of \Cref{fig:hoare:while-lb}, in which
$\bowtie^{-1}$ is the converse relation:
${\leq}^{-1}={\geq}$, ${\geq}^{-1}={\leq}$ and ${=}^{-1}={=}$.

\paragraph{Semantic facts.}
We use the following four facts:
\begin{align}
  \neg B&\models V=g,
  \label{eq:cook-loop-V-notB}\\
  B&\models V=\WP{D}{V},
  \label{eq:cook-loop-V-B}\\
  \neg B\land P&\models Q,
  \label{eq:cook-loop-P-notB}\\
  B\land P&\models\WP{D}{P}.
  \label{eq:cook-loop-P-B}
\end{align}

\noindent\emph{Proof of \eqref{eq:cook-loop-V-notB}.}
Let $\neg B(m)$.  By \Cref{lem:In-recursion}, $I_0(m)=g(m)$ and
$I_{n+1}(m)=g(m)$, so $I_n(m)=g(m)$ for every $n$.  Hence
$V(m)=\sup_n I_n(m)=g(m)$, again by \Cref{lem:In-recursion}.

\noindent\emph{Proof of \eqref{eq:cook-loop-V-B}.}
Let $B(m)$.  Then
\[
  \begin{aligned}
  V(m)
  &\overset{\text{\Cref{lem:In-recursion}}}{=}
  \sup_n I_{n+1}(m)
  \overset{\text{\Cref{lem:In-recursion}}}{=}
  \sup_n\WP{D}{I_n}(m)\\
  &\overset{\text{MC}}{=}
  \WP{D}{\sup_n I_n}(m)
  \overset{\text{\Cref{lem:In-recursion}}}{=}
  \WP{D}{V}(m).
  \end{aligned}
\]

\noindent\emph{Proof of \eqref{eq:cook-loop-P-notB}.}
Let $\neg B(m)$ and $P(m)$.  Then
\[
  \begin{aligned}
  P(m)
  &\iff
  \WP{L}{Q}(m)
  \overset{\eqref{eq:cook-gwp-definition}}{\iff}
  \forall m'\in\supp(\llbracket L\rrbracket(m)).\ Q(m')\\
  &\overset{\eqref{eq:cook-denote-loop-functional},\eqref{eq:cook-denote-loop}}{\iff}
  \forall m'\in\supp(\delta_m).\ Q(m')
  \iff
  Q(m).
  \end{aligned}
\]

\noindent\emph{Proof of \eqref{eq:cook-loop-P-B}.}
Let $B(m)$ and $P(m)$, and let $m''\in\supp(\llbracket D\rrbracket(m))$.  We
show $P(m'')$, i.e.\ that every $m'\in\supp(\llbracket L\rrbracket(m''))$
satisfies $Q$.  Take such an $m'$.  By \eqref{eq:cook-denote-loop},
$W_n(m'')(m')>0$ for some $n$.  By \eqref{eq:cook-denote-loop-functional} and
\eqref{eq:cook-bind-def},
\[
  W_{n+1}(m)(m')
  =\sum_{x\in\supp(\llbracket D\rrbracket(m))}
    \llbracket D\rrbracket(m)(x)\cdot W_n(x)(m')
  \geq\llbracket D\rrbracket(m)(m'')\cdot W_n(m'')(m')
  >0.
\]
By \eqref{eq:cook-denote-loop},
$\llbracket L\rrbracket(m)(m')\geq W_{n+1}(m)(m')>0$, so
$m'\in\supp(\llbracket L\rrbracket(m))$, and $Q(m')$ follows from $P(m)$ by
\eqref{eq:cook-gwp-definition}.  As $m''$ was arbitrary,
\eqref{eq:cook-gwp-definition} gives $\WP{D}{P}(m)$.

\paragraph{Premise 1.}
The IH for $D$, at postcondition $P$, post-expectation $V$ and relation
$\bowtie$, is
\begin{equation}
  \vdash\HT{\WP{D}{P}}{\WP{D}{V}}{D}{P}{V}.
  \label{eq:cook-loop-body-IH}
\end{equation}
\textsc{Consequence} turns it into Premise~1 of \textsc{While}:
\begin{equation}
\frac{
  \begin{array}{c}
    B\land P\models\WP{D}{P}\land V\bowtie\WP{D}{V}
    \\[1mm]
    \qquad
    \vdash\HT{\WP{D}{P}}{\WP{D}{V}}{D}{P}{V}
    \\[1mm]
    P\models P\land V\bowtie V
  \end{array}
}{
  \vdash\HT{B\land P}{V}{D}{P}{V}
}\;\textsc{Consequence}.
\label{eq:cook-loop-premise1}
\end{equation}
The first premise follows from \eqref{eq:cook-loop-P-B} and
\eqref{eq:cook-loop-V-B}, since equality implies $\bowtie$; the middle premise
is the IH \eqref{eq:cook-loop-body-IH}; the last premise is reflexive.

The remaining premises depend on the choice of the witnesses $r_i$, which
depends on $\bowtie$.

\paragraph{Premises 2--4 for $\bowtie={\geq}$.}
Here $\bowtie^{-1}={\leq}$.  Take $r_i\defsym0$ for every $i$.
\begin{itemize}
  \item Premise~2, $B\land P\models V\geq0$, holds because $V$ is
    non-negative.
  \item Premise~4, $B\land P\models\lim_i r_i=0$, holds trivially.
  \item Premise~3 asks for
    $\vdash\HT[lower]{B\land P}{0}{D}{\top}{0}$ for every $i$.  The IH for
    $D$, at postcondition $\top$, post-expectation $0$ and relation $\leq$, is
    $\vdash\HT[lower]{\WP{D}{\top}}{\WP{D}{0}}{D}{\top}{0}$, and
    \[
    \frac{
      \begin{array}{c}
        B\land P\models\WP{D}{\top}\land 0\leq\WP{D}{0}
        \\[1mm]
        \qquad
        \vdash\HT[lower]{\WP{D}{\top}}{\WP{D}{0}}{D}{\top}{0}
        \\[1mm]
        \top\models\top\land 0\leq 0
      \end{array}
    }{
      \vdash\HT[lower]{B\land P}{0}{D}{\top}{0}
    }\;\textsc{Consequence}.
    \]
    The first premise holds because $\WP{D}{\top}=\top$ by
    \eqref{eq:cook-gwp-definition} and $\WP{D}{0}=0$ by
    \eqref{eq:cook-wp-definition}; the last premise is reflexive.
\end{itemize}

\paragraph{Premises 2--4 for $\bowtie\in\{\leq,=\}$.}
Here $\bowtie^{-1}\in\{\geq,=\}$, and we use that $V$ is finite-valued.
Define
\begin{equation}
  K_0\defsym0,
  \qquad
  K_{n+1}\defsym I_n,
  \qquad
  r_n\defsym V-K_n.
  \label{eq:cook-residual-definition}
\end{equation}
The $K_n$ increase with $n$ and, by \Cref{lem:In-recursion},
$\sup_n K_n=\sup_n I_n=V$.  Hence $0\leq K_n\leq V$, and each $r_n$ is a
well-defined, non-negative, finite-valued expectation: $r_n$ is the part of
$V$ contributed by runs that execute the body at least $n$ times.

\noindent\emph{Premise 2.}
By \eqref{eq:cook-residual-definition}, $r_0=V-0=V$, so $B\land P\models V=r_0$,
and equality implies $\bowtie$.

\noindent\emph{Premise 4.}
Since $K_n$ increases to $V$ and $V$ is finite,
$\lim_n r_n=V-\lim_n K_n=V-V=0$ on every memory, in particular on $B\land P$.

\noindent\emph{The residual equation.}
For every $n$,
\begin{equation}
  B\models r_{n+1}=\WP{D}{r_n}.
  \label{eq:cook-residual-step}
\end{equation}
First, $B\models K_{n+1}=\WP{D}{K_n}$.  For $n=0$, \Cref{lem:In-recursion}
gives $K_1=I_0=[\neg B]\cdot g$, which is $0$ on $B$, and
$\WP{D}{K_0}=\WP{D}{0}=0$.  For $n\geq1$, on $B$,
\Cref{lem:In-recursion} gives $K_{n+1}=I_n=\WP{D}{I_{n-1}}=\WP{D}{K_n}$.
Now, on $B$,
\[
  K_{n+1}+r_{n+1}
  \overset{\eqref{eq:cook-residual-definition}}{=}
  V
  \overset{\eqref{eq:cook-loop-V-B}}{=}
  \WP{D}{V}
  \overset{\eqref{eq:cook-residual-definition}}{=}
  \WP{D}{K_n+r_n}
  =
  \WP{D}{K_n}+\WP{D}{r_n}
  =
  K_{n+1}+\WP{D}{r_n}.
\]
The fourth equality is linearity of $\mathsf{wp}$ (by
\eqref{eq:cook-wp-definition} and \eqref{eq:cook-expect-def}, $\mathsf{wp}$ is
a sum), and the last one is $B\models K_{n+1}=\WP{D}{K_n}$.  Since
$K_{n+1}\leq V$ is finite, it cancels, which gives
\eqref{eq:cook-residual-step}.

\noindent\emph{Premise 3.}
Fix $n$.  The lemma is proved for all relations at once, so the IH for $D$
can be used at relation $\bowtie^{-1}$, postcondition $\top$ and
post-expectation $r_n$:
\begin{equation}
  \vdash\HT[\rel^{-1}]{\WP{D}{\top}}{\WP{D}{r_n}}{D}{\top}{r_n}.
  \label{eq:cook-loop-residual-IH}
\end{equation}
\textsc{Consequence} at relation $\bowtie^{-1}$ gives Premise~3:
\begin{equation}
\frac{
  \begin{array}{c}
    B\land P\models\WP{D}{\top}\land r_{n+1}\mathrel{\bowtie^{-1}}\WP{D}{r_n}
    \\[1mm]
    \qquad
    \vdash\HT[\rel^{-1}]{\WP{D}{\top}}{\WP{D}{r_n}}{D}{\top}{r_n}
    \\[1mm]
    \top\models\top\land r_n\mathrel{\bowtie^{-1}}r_n
  \end{array}
}{
  \vdash\HT[\rel^{-1}]{B\land P}{r_{n+1}}{D}{\top}{r_n}
}\;\textsc{Consequence}.
\label{eq:cook-loop-residual-conseq}
\end{equation}
The first premise holds because $\WP{D}{\top}=\top$ by
\eqref{eq:cook-gwp-definition} and because of the residual equation
\eqref{eq:cook-residual-step}, since equality implies $\bowtie^{-1}$; the
middle premise is the IH \eqref{eq:cook-loop-residual-IH}; the last premise is
reflexive.

\paragraph{Application of \textsc{While}.}
In both branches all four premises hold, so the \textsc{While} rule gives
\begin{equation}
\frac{
  \begin{array}{c}
    \vdash\HT{B\land P}{V}{D}{P}{V}
    \qquad
    B\land P\models V\bowtie r_0
    \\[1mm]
    \forall i\geq0.\ \vdash\HT[\rel^{-1}]{B\land P}{r_{i+1}}{D}{\top}{r_i}
    \qquad
    B\land P\models\lim_{i\to\infty}r_i=0
  \end{array}
}{
  \vdash\HT{P}{V}{L}{\neg B\land P}{V}
}\;\textsc{While}.
\label{eq:cook-loop-while}
\end{equation}

\paragraph{The final consequence step.}
\begin{equation}
\frac{
  P\models P\land V\bowtie V
  \qquad
  \vdash\HT{P}{V}{L}{\neg B\land P}{V}
  \qquad
  \neg B\land P\models Q\land V\bowtie g
}{
  \vdash\HT{P}{V}{L}{Q}{g}
}\;\textsc{Consequence}.
\label{eq:cook-loop-final-conseq}
\end{equation}
The first premise is reflexive, the middle premise is
\eqref{eq:cook-loop-while}, and the last premise follows from
\eqref{eq:cook-loop-P-notB} and \eqref{eq:cook-loop-V-notB}, since equality
implies $\bowtie$.  Since $P=\WP{L}{Q}$ and $V=\WP{L}{g}$, the conclusion is
the exact-pre judgment \eqref{eq:cook-exact-bowtie} for $L$.
\end{proof}

\subsubsection{From the exact pre-expectation to every valid judgment}

\begin{theorem}[Completeness]
\label{thm:cook-relative-completeness}
Under the assumptions of this section, for every command $C$, predicates
$P,Q$, expectations $f,g$ and $\bowtie\in\{\leq,=,\geq\}$,
\[
  \valid\HT{P}{f}{C}{Q}{g}
  \quad\Longrightarrow\quad
  \vdash\HT{P}{f}{C}{Q}{g}.
\]
\end{theorem}

\begin{proof}
Assume $\valid\HT{P}{f}{C}{Q}{g}$.  By
\eqref{eq:cook-general-validity-entailment},
$P\models\WP{C}{Q}\land f\bowtie\WP{C}{g}$.  By
\Cref{lem:exact-pre-derivable},
$\vdash\HT{\WP{C}{Q}}{\WP{C}{g}}{C}{Q}{g}$.  These are the first two premises
of the \textsc{Consequence} instance
\eqref{eq:cook-general-final-consequence-outline}; its third premise,
$Q\models Q\land g\bowtie g$, is reflexive.  Its conclusion is
$\vdash\HT{P}{f}{C}{Q}{g}$.
\end{proof}
